"""Archive audited stock NCU text/metadata, never model or tensor payloads."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-stock-dot-ncu-evidence-20261001"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    destination = ROOT / "results/a3-stock-dot-ncu-archive-20261001"
    destination.mkdir(exist_ok=False)
    audit = json.loads((ROOT / "results/a3-stock-dot-ncu-cpu-audit-20261001-r1/result.json").read_text())
    assert audit["stage"] == "PASS_CPU_STOCK_NCU_AND_OUTPUT_DTYPE_AUDIT_NOT_MODEL_SUCCESS"
    selected = {}
    for name in (
        "a3-stock-dot-ncu-20261001-r1",
        "a3-stock-dot-ncu-20261001-r2",
        "a3-stock-dot-ncu-20261001-r3",
        "a3-stock-dot-ncu-analysis-20261001-r3",
        "a3-bf16-dot-fp32-output-20261001-r1",
        "a3-stock-dot-ncu-cpu-audit-20261001-r1",
    ):
        for directory in (ROOT / "results" / name, ROOT / "results" / (name + "-launch")):
            if directory.exists():
                for path in sorted(directory.rglob("*")):
                    if path.is_file() and path.suffix in (".json", ".log", ".txt", ".csv"):
                        selected[str(path.relative_to(ROOT))] = path
    for name in (
        "run_a3_stock_dot_ncu_queued_20261001.py",
        "run_a3_stock_dot_ncu_clocknone_queued_20261001.py",
        "run_a3_stock_dot_ncu_clocknone_readlock_queued_20261001.py",
        "a3_stock_dot_ncu_probe_20261001.py",
        "diagnose_a3_bf16_dot_fp32_output_queued_20261001.py",
        "audit_a3_stock_dot_ncu_20261001.py",
        "audit_a3_precision_followup_r2_20261001.py",
        "archive_a3_stock_dot_ncu_20261001.py",
    ):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    records = [{"path": name, "size": path.stat().st_size, "sha256": sha(path)} for name, path in sorted(selected.items())]
    manifest = {
        "stage": audit["stage"],
        "files": records,
        "excluded": ["models", "cache", "engines", ".pt outputs", "native binaries", "binary ncu-rep kept on Thor with recorded SHA"],
        "scope": "Original first-expert stock BF16 dot NCU plus BF16-input/FP32-output control, independently CPU audited; no native/model/performance pass",
    }
    payload = (json.dumps(manifest, indent=2) + "\n").encode()
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for name, path in selected.items():
            tar.add(path, arcname=PREFIX + "/" + name, recursive=False)
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    transport = {"archive": str(archive), "sha256": sha(archive), "size": archive.stat().st_size, "files": len(records) + 1}
    (destination / "transport.json").write_text(json.dumps(transport, indent=2) + "\n")
    print(json.dumps(transport), flush=True)


if __name__ == "__main__":
    main()
