"""Frozen production-arithmetic candidate regression, admitted before Torch imports."""

import argparse
import fcntl
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time
import xml.etree.ElementTree as ET

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
HEAD = "741e10ffa5e93dbc0a8a41e06e66e0c81c1f85b1"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    source = ROOT / "a3sfdlateste2"
    python = ROOT / "venvs/a3sfdlateste2/bin/python"
    names = [
        "a3_production_optional_c_store_native_kernel_20261001.py",
        "a3_production_c_store_pytest_plugin_20261001.py",
        "test_a3_production_c_store_pair_r2_20261001.py",
        "a3_repair_matrix_select_20261001.py",
    ]
    frozen = {name: sha(ROOT / "tools" / name) for name in names}
    with open("/tmp/codex-thor-perf.lock", "a") as lock:
        start = time.monotonic()
        (out / "admission_wait.json").write_text(json.dumps({"stage": "WAITING_FOR_SHARED_LOCK", "pid": os.getpid(), "torch_imported": False}))
        while True:
            try:
                fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                break
            except BlockingIOError:
                if time.monotonic() - start > 3600:
                    return 75
                time.sleep(1)
        assert "torch" not in sys.modules
        (out / "admission.json").write_text(
            json.dumps({"stage": "LOCK_ADMITTED", "pid": os.getpid(), "torch_imported": False, "wait_seconds": time.monotonic() - start})
        )
        assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
        assert subprocess.check_output(["git", "-C", str(source), "rev-parse", "HEAD"], text=True).strip() == HEAD
        core = [
            "gemm/cutedsl/test_grouped_swiglu_sfd_tiles.py",
            "gemm/cutedsl/test_swiglu_amax_per_call.py",
            "gemm/cutedsl/test_swiglu_wrapper_launch_stream.py",
        ]
        extra = [
            "gemm/cutedsl/test_grouped_gemm_swiglu.py::test_grouped_gemm_swiglu_wrapper_fp4",
            "gemm/cutedsl/test_grouped_gemm_swiglu.py::test_grouped_gemm_swiglu_wrapper_fp8",
            "gemm/cutedsl/test_grouped_gemm_canonical_layouts.py::test_grouped_gemm_swiglu_canonical_matches_legacy",
            "gemm/cutedsl/test_grouped_gemm_canonical_layouts.py::test_grouped_gemm_swiglu_canonical_bf16_prob",
            str(ROOT / "tools/test_a3_production_c_store_pair_r2_20261001.py"),
        ]
        results = []
        for mode, tests, expected in (("emit", core + extra, 63), ("omit", core, 16)):
            env = dict(
                os.environ,
                PYTHONPATH=str(ROOT / "tools") + ":" + str(source / "test/python"),
                A3_EXPECT_SOURCE=str(source),
                A3_IMPORT_LOG=str(out / mode / "imports"),
                A3_C_STORE_MODE=mode,
            )
            logger = "/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py"
            command = [
                str(python),
                logger,
                "--out",
                str(out / mode / "pytest"),
                "--cwd",
                str(source / "test/python"),
                "--timeout",
                "1800",
                "--",
                str(python),
                "-m",
                "pytest",
                "-p",
                "a3_production_c_store_pytest_plugin_20261001",
                "-p",
                "a3_repair_matrix_select_20261001",
                "-o",
                "addopts=",
                "-q",
                "--grouped-gemm-nkl=1024,1024,4",
                "--junitxml=" + str(out / mode / "pytest/junit.xml"),
                *tests,
            ]
            rc = subprocess.run(command, env=env).returncode
            if rc:
                return rc
            run = json.loads((out / mode / "pytest/run.json").read_text())
            xml = ET.parse(out / mode / "pytest/junit.xml").getroot()
            cases = list(xml.iter("testcase"))
            assert len(cases) == expected and not any(list(xml.iter(name)) for name in ("skipped", "failure", "error"))
            records = [json.loads(p.read_text()) for p in (out / mode / "imports").glob("*.json") if not p.name.startswith("matrix-selection-")]
            workers = [v for v in records if v["worker"] is not None]
            assert workers and all(v["pytest_exitstatus"] == 0 and v["python_execute_calls_including_capture"] > 0 for v in workers)
            assert all(v["source_head"] == HEAD and v["pytest_exitstatus"] == 0 for v in records)
            results.append({"mode": mode, "passed": expected, "skipped": 0, "run": run, "imports": records})
        assert frozen == {name: sha(ROOT / "tools" / name) for name in names}
        assert not subprocess.check_output(["git", "-C", str(source), "status", "--porcelain", "--untracked-files=no"], text=True).strip()
        assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
        result = {
            "stage": "PRIVATE_PRODUCTION_C_STORE_REGRESSION_PASS_NOT_MODEL_SUCCESS",
            "head": HEAD,
            "results": results,
            "frozen_sources": frozen,
            "harness_sha256": sha(__file__),
            "service_final": "inactive",
            "production_source_modified": False,
            "feature_accepted": False,
            "formal_model_performance": "NOT_RUN",
            "limits": "Original production FP32 activation/support/stream semantics, private kernel classes outside source. Emit31 existing +32 three-arm exact geometry/vector/dtype comparisons; omit16 existing SFD/AMAX/reuse/graph/stream cases. No public flag/cache-key/API contract, no model accuracy pass or model throughput claim.",
        }
        (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
        print(json.dumps({"stage": result["stage"], "passed": 79}), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
