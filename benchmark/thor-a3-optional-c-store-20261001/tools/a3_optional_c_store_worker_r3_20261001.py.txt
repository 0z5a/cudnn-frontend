"""Full24 optional-C-write private prototype parity and captured execution timing.

Same original checkpoint/all24/E32/top8/H1024/I512 and original baseline/gates.
Actual MXFP8 tensor-core native D is consumed. Explicit experimental kernel
rounds C, SiLU and product to BF16 before SFD normalization. Frozen independent
oracle recasts reconstructed logical operands to BF16, applies stock BF16 dot,
and BF16 SiLU/product. This declared numerical recipe is fixed before tests;
it is not a replacement for failed native execution. K32 native dot order is retained; the only new execution configuration is
Omit intermediate C stores, versus emit C stores. C output is explicitly
unavailable in the private omit mode; its workspace is still allocated by
the unmodified wrapper. This is not a public wrapper/API contract change.
This can save write traffic but does not save allocation or shared memory.
Both modes explicitly acquire D store pipeline after commit, before SMEM reuse.
Prior failed loader and no-C/D-TMA-reuse evidence is preserved separately. The independent oracle is unchanged. No native BF16
matrix multiplication, fallback, support bypass or production edit.
"""

def expanded_quantize(value, operand):
    high, _, scale_high = quantize(value)
    low, _, scale_low = quantize(value.float() - m.dequantize(high, scale_high))
    q_parts, s_parts = (
        ((high, high, low, low), (scale_high, scale_high, scale_low, scale_low))
        if operand == "A"
        else (
            (high, low, high, low),
            (scale_high, scale_low, scale_high, scale_low),
        )
    )
    q = t.cat([x.view(t.uint8) for x in q_parts], dim=-1).view(t.float8_e4m3fn)
    scale = t.cat(s_parts, dim=-1)
    # Component-major -> originalK32-block-major, with matched scale groups.
    permutation = t.arange(4096, device=q.device).reshape(4, 32, 32).permute(1, 0, 2).reshape(-1) if K_BLOCK else t.arange(4096, device=q.device)
    sf_permutation = permutation.reshape(-1, 32)[:, 0] // 32
    q = q.view(t.uint8).index_select(-1, permutation).view(t.float8_e4m3fn)
    scale = scale.index_select(-1, sf_permutation)
    rows, cols = q.shape[-2:]
    assert cols == 4096
    codes = (t.log2(scale).int() + 127).to(t.uint8)
    raw = codes.reshape(-1, rows // 128, 4, 32, cols // 128, 4)
    packed = raw.permute(0, 1, 4, 3, 2, 5).contiguous().view(t.float8_e8m0fnu)
    effective = m.dequantize(high, scale_high) + m.dequantize(low, scale_low)
    return q, packed, scale, effective


def make_adapter(base, expert, layer, bf16_stage=False):
    class DoubleAdapter(base):
        def __init__(self):
            super().__init__(expert, layer, (256, 256))
            gate, up = expert.gate_up_proj.detach().chunk(2, 1)
            weight = t.stack((gate.reshape(32, -1, 32, 1024), up.reshape(32, -1, 32, 1024)), 2).reshape(32, 1024, 1024)
            self.b, self.sfb, self.bscale, effective = expanded_quantize(weight, "B")
            self.weight_error = m.difference(effective, weight)
            self.bref = effective.to(t.bfloat16)
            if isinstance(self, p.PrecisionAdapter):
                self.policy = "merge_fp32"

        def reference_staged(self, hidden, index, probability):
            x, ends, destinations, order, token_ids, sizes, aligned = self.routing(hidden, index)
            a, sfa, scale, effective = expanded_quantize(x, "A")
            adeq = effective.to(t.bfloat16)
            if self.bref is None:
                self.bref = m.dequantize(self.b, self.bscale)
            d = t.zeros((x.shape[0], self.i), dtype=t.bfloat16, device=x.device)
            start = 0
            for expert, n in enumerate(aligned):
                if n:
                    c = m.F.linear(adeq[start : start + n], self.bref[expert]).to(t.bfloat16).reshape(n, -1, 2, 32)
                    d[start : start + n] = (m.F.silu(c[:, :, 0]) * c[:, :, 1]).reshape(n, self.i)
                start += n
            live = d[destinations]
            routes = t.empty((order.numel(), self.h), dtype=t.bfloat16, device=x.device)
            weight = probability.flatten()[order]
            start = 0
            for expert, n in enumerate(sizes):
                if n:
                    y = m.F.linear(live[start : start + n], self.expert.down_proj[expert])
                    routes[start : start + n] = (y * weight[start : start + n, None]).to(t.bfloat16)
                start += n
            unsorted = t.empty_like(routes)
            unsorted[order] = routes
            return unsorted.reshape(hidden.shape[0], index.shape[-1], self.h).sum(1)

        def forward(self, *args):
            if bf16_stage:
                if self.mode == "reference":
                    return self.reference_staged(*args)
            assert m.quantize_mxfp8 is quantize
            m.quantize_mxfp8 = lambda x: expanded_quantize(x, "A")[:3]
            try:
                return super().forward(*args)
            finally:
                m.quantize_mxfp8 = quantize

    return DoubleAdapter()


def compatibility(oracle, stock):
    r, b = oracle.double(), stock.double()
    error = (t.linalg.vector_norm(r - b) / t.linalg.vector_norm(b)).item()
    ratio = (t.linalg.vector_norm(r) / t.linalg.vector_norm(b)).item()
    bound = max(0.0, error - 0.01 * ratio)
    return {"oracle_bf16_relative_l2_fp64": error, "norm_ratio_fp64": ratio, "native_bf16_lower_bound": bound, "both_gates_impossible": bound > 0.05}

    model = model.eval().cuda()
    experts = [e for e in model.modules() if isinstance(e, m.GraniteMoeExperts)]
    assert len(experts) == 24
    tokenizer = m.AutoTokenizer.from_pretrained(m.MODEL, local_files_only=True, trust_remote_code=False)
    _, prompts, workloads = m.load_inputs(tokenizer)
    cases = [prompts[0], prompts[1], workloads["256"]]
    baseline_dir = m.ROOT / "results/a3-model-repair-a3sfdlatest4-n256-20261001"
    baseline_meta = json.loads((baseline_dir / "result.json").read_text())
    assert baseline_meta["source_head"] == "7bfeda6bfeaa63a1e59fadc5d9fb48676f03f744"
    baseline = t.load(baseline_dir / "outputs.pt", map_location="cpu", weights_only=True)
    paired_dir = m.ROOT / "results/a3-mxfp8-bf16-staged-full-model-20261001-r1"
    paired_meta = json.loads((paired_dir / "result.json").read_text())
    paired_run = json.loads((paired_dir.with_name(paired_dir.name + "-launch") / "run.json").read_text())
    assert paired_run["status"] == "EXITED" and paired_run["returncode"] == 0
    assert (
        paired_meta["source_head"] == args.expected_head
        and paired_meta["experimental_native_kernel"]["sha256"] == "68d7c926dec0fbdf7bb0015a80df605d4ba8045af83929db5376b3b66b89feb3"

    assert paired_meta["imports"]["binding"]["sha256"] == m.sha_file(m._compiled_module.__file__)
    paired = t.load(paired_dir / "outputs.pt", map_location="cpu", weights_only=True)
    for name, k_block in (("k32_emit_c", 32), ("k32_omit_c", 32)):
        m.api.BlockScaledContiguousGroupedGemmKernel = EmitC if name == "k32_emit_c" else OmitC
        m.api._cache_of_GroupedGemmSwigluSm100Objects = {}
        K_BLOCK = k_block
        base, bf16_stage = m.A3ExpertAdapter, True
        adapters = [make_adapter(base, expert, layer, bf16_stage) for layer, expert in enumerate(experts)]
        policies.append(
            {
                "policy": name,
                "weight_error_max_l2": max(a.weight_error["relative_l2"] for a in adapters),
                "weight_error_max_abs": max(a.weight_error["max_abs"] for a in adapters),
                "native_semantics_supported": True,
            }
        )
        with t.inference_mode():
            for case, ids in enumerate(cases):
                m.attach(adapters, "reference")
                before = m.NATIVE_CALLS
                oracle = model(input_ids=ids.cuda(), use_cache=False).logits.cpu()
                assert m.NATIVE_CALLS == before
                saved[f"{name}/{case}/reference"] = oracle
                compare = compatibility(oracle, baseline[f"forward_{case}_stock"])

                    row.update(

        try:
