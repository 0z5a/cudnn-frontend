"""Archive terminal C-store diagnostics, failed harnesses and verified private tests."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-optional-c-store-evidence-20261001"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    runs = {
        "a3-optional-c-store-20261001-r1": 1,
        "a3-optional-c-store-20261001-r2": 1,
        "a3-optional-c-store-20261001-r3": 0,
        "a3-c-store-original-control-20261001-r1": 0,
        "a3-optional-c-store-cpu-audit-20261001-r1": 0,
        "a3-production-c-store-regression-20261001-r1": 1,
        "a3-production-c-store-regression-20261001-r2": 0,
        "a3-production-c-store-three-arm-20261001-r1": 0,
        "a3-production-c-store-cpu-audit-20261001-r1": 0,
    }
    selected = {}
    for name, expected in runs.items():
        directory = ROOT / "results" / name
        launch = directory.with_name(name + "-launch")
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["signal"] is None and run["returncode"] == expected
        for base in (directory, launch):
            for path in sorted(base.rglob("*")):
                if path.is_file() and path.suffix in (".json", ".log", ".xml", ".txt"):
                    selected[str(path.relative_to(ROOT))] = path
    for name, stage in (
        ("a3-optional-c-store-cpu-audit-20261001-r1", "CPU_AUDITED_C_STORE_PARITY_AND_THREE_ARM_CAPTURED_TIMING_NOT_MODEL_SUCCESS"),
        ("a3-production-c-store-cpu-audit-20261001-r1", "CPU_AUDITED_PRIVATE_PRODUCTION_C_STORE_79_TESTS_54_TENSORS_NOT_MODEL_SUCCESS"),
    ):
        assert json.loads((ROOT / "results" / name / "result.json").read_text())["stage"] == stage
    for name in (
        "a3_optional_c_store_native_kernel_20261001.py",
        "a3_optional_c_store_native_kernel_r3_20261001.py",
        "a3_optional_c_store_worker_20261001.py",
        "a3_optional_c_store_worker_r3_20261001.py",
        "a3_optional_c_store_queued_20261001.py",
        "a3_optional_c_store_queued_r2_20261001.py",
        "a3_optional_c_store_queued_r3_20261001.py",
        "a3_c_store_original_control_queued_20261001.py",
        "audit_a3_optional_c_store_20261001.py",
        "a3_production_optional_c_store_native_kernel_20261001.py",
        "a3_production_c_store_pytest_plugin_20261001.py",
        "test_a3_production_c_store_pair_20261001.py",
        "test_a3_production_c_store_pair_r2_20261001.py",
        "run_a3_production_c_store_regression_queued_20261001.py",
        "run_a3_production_c_store_regression_queued_r2_20261001.py",
        "a3_production_c_store_three_arm_queued_20261001.py",
        "audit_a3_production_c_store_20261001.py",
        "audit_a3_production_c_store_r2_20261001.py",
        "archive_a3_optional_c_store_20261001.py",
        "a3_mxfp8_bf16_staged_native_kernel_20261001.py",
        "a3_all24_precision_probe_20261001.py",
        "a3_model_e2e_mergefix_20261001.py",
        "audit_a3_precision_followup_r2_20261001.py",
        "a3_repair_matrix_select_20261001.py",
    ):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    for name in (
        "a3-latest-e2-native-provenance-20261001.json",
        "a3-mxfp8-bf16-staged-full-model-20261001-r1/result.json",
        "a3-n128-mxfp8-staged-trace-cpu-audit-20261001-r1/result.json",
    ):
        path = ROOT / "results" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    source = ROOT / "a3sfdlateste2"
    for relative in (
        "python/cudnn/gemm/cutedsl/grouped/swiglu/grouped_gemm_swiglu_quant.py",
        "python/cudnn/gemm/cutedsl/grouped/swiglu/api.py",
        "test/python/gemm/cutedsl/test_grouped_swiglu_sfd_tiles.py",
        "test/python/gemm/cutedsl/test_swiglu_amax_per_call.py",
        "test/python/gemm/cutedsl/test_swiglu_wrapper_launch_stream.py",
        "test/python/gemm/cutedsl/test_grouped_gemm_swiglu.py",
        "test/python/gemm/cutedsl/test_grouped_gemm_canonical_layouts.py",
    ):
        selected["runtime/" + relative] = source / relative
    manifest = {
        "stage": "PRIVATE_C_STORE_EVIDENCE_CPU_AUDITED_NOT_FULL_MODEL_SUCCESS",
        "scope": "156 saved tensor entries independentlyCPUaudited; all6 experimental fullmodel native results byteequal frozencontrol but qualityfails. ExplicitD-TMA protection repairs naiveomitC corruption. 79 production-arithmetic GPUtests pass in corrected harness; failed loader, unsafeomitC and singleton-stride CPUcomparison histories preserved. Three-arm balanced capturedwrapper timing, no publicAPI acceptance/modelthroughput or allocation saving.",
        "runs": runs,
        "files": [{"path": name, "size": path.stat().st_size, "sha256": sha(path)} for name, path in sorted(selected.items())],
        "excluded": ["models", "cache", "engines", ".pt outputs", "native binaries"],
    }
    destination = ROOT / "results/a3-optional-c-store-archive-20261001"
    destination.mkdir(exist_ok=False)
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for name, path in selected.items():
            tar.add(path, arcname=PREFIX + "/" + name, recursive=False)
        payload = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    record = {"archive": str(archive), "sha256": sha(archive), "size": archive.stat().st_size, "files": len(selected) + 1}
    (destination / "transport.json").write_text(json.dumps(record, indent=2) + "\n")
    print(json.dumps(record), flush=True)


if __name__ == "__main__":
    main()
