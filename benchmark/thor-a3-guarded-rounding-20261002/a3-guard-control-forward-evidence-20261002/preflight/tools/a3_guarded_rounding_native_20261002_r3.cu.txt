// Private numerical control: actual MXFP8 TC C is retained when certified.
// Ambiguous outputs are recomputed from FP8 codes/scales using integer ALU.
// No BF16 Tensor Core GEMM, Torch dot, model-oracle input or fitted bias.

__device__ int
exponent64(double x) {
    return x == 0.0 ? -10000 : int((uint64_t(__double_as_longlong(x)) >> 52) & 2047) - 1023;
}

__device__ double
power64(int exponent) {
    return __longlong_as_double(int64_t(exponent + 1023) << 52);
}

__device__ double
error_bound(double s, int product_raw, int group, int updates) {
    if (s == 0.0) return 0.0;
    double error = 0.0;
    for (int i = 0; i < updates; ++i) {
        int quantum         = max(exponent64(s + error), product_raw) - 25;
        double alignment    = (group + 1) * power64(quantum);
        double intermediate = s + error + alignment;
        double ulp          = power64(max(exponent64(intermediate) - 23, -149));
        error               = error + alignment + ulp;
    }
    return error * (1.0 + 16.0 * updates * 0x1p-53);
}

__device__ uint16_t
bfword(float x) {
    return __bfloat16_as_ushort(__float2bfloat16_rn(x));
}

__device__ bool
certify(float value, const double *xs, const double *ws) {
    double native_s       = xs[1] * ws[1];
    double target_s       = xs[0] * ws[0];
    double en             = error_bound(native_s, int(xs[4]) + int(ws[4]), 32, 128);
    double er             = error_bound(target_s, int(xs[3]) + int(ws[3]), 16, 64);
    double reconstruction = xs[2] * ws[0] + xs[0] * ws[2] + xs[2] * ws[2];
    double bound          = (en + er + reconstruction) * (1.0 + 64.0 * 0x1p-53);
    double low = double(value) - bound, high = double(value) + bound;
    float lower = __double2float_rn(low), upper = __double2float_rn(high);
    if (double(lower) > low) lower = nextafterf(lower, -CUDART_INF_F);
    if (double(upper) < high) upper = nextafterf(upper, CUDART_INF_F);
    return bfword(lower) == bfword(upper);
}

__device__ int64_t
sfindex(int row, int scale_column, int rows, int batch) {
    // Physical F8_128x4: [batch,row/128,K/128,row%32,(row%128)/32,Kscale%4].
    return (((((int64_t(batch) * (rows / 128) + row / 128) * 32 + scale_column / 4) * 32 + row % 32) * 4 +
             (row % 128) / 32) *
                4 +
            scale_column % 4);
}

__device__ float
decode_scaled(uint8_t code, uint8_t scale) {
    int encoded  = (code >> 3) & 15;
    int mantissa = (code & 7) + (encoded ? 8 : 0);
    if (code & 128) mantissa = -mantissa;
    return ldexpf(float(mantissa), max(encoded, 1) - 10 + int(scale) - 127);
}

__device__ uint16_t
reconstructed_word(const uint8_t *codes, const uint8_t *scales, int row, int k, int rows, int batch, bool operand_a) {
    int block = k / 32, lane = k % 32, low_component = operand_a ? 2 : 1;
    int64_t base = (int64_t(batch) * rows + row) * 4096 + block * 128;
    float high   = decode_scaled(codes[base + lane], scales[sfindex(row, block * 4, rows, batch)]);
    float low    = decode_scaled(codes[base + low_component * 32 + lane],
                              scales[sfindex(row, block * 4 + low_component, rows, batch)]);
    return bfword(__fadd_rn(high, low));
}

__device__ void
decode_bf(uint16_t word, int &mantissa, int &raw) {
    int encoded = (word >> 7) & 255;
    mantissa    = (word & 127) + (encoded ? 128 : 0);
    if (word & 32768) mantissa = -mantissa;
    raw = max(encoded, 1) - 127;
}

__device__ int64_t
align_integer(int64_t mantissa, int exponent, int quantum) {
    if (!mantissa) return 0;
    int shift          = exponent - quantum;
    uint64_t magnitude = mantissa < 0 ? uint64_t(-mantissa) : uint64_t(mantissa);
    magnitude          = shift >= 0 ? magnitude << shift : (-shift >= 64 ? 0 : magnitude >> -shift);
    return mantissa < 0 ? -int64_t(magnitude) : int64_t(magnitude);
}

__device__ uint32_t
normalize_rtz(int64_t value, int quantum) {
    if (!value) return 0;
    uint32_t sign      = value < 0 ? 0x80000000u : 0;
    uint64_t magnitude = value < 0 ? uint64_t(-value) : uint64_t(value);
    int length         = 64 - __clzll(magnitude);
    int exponent       = quantum + length - 1;
    if (exponent > 127) return sign | 0x7f800000u;
    if (exponent < -126) {
        int shift         = quantum + 149;
        uint64_t fraction = shift >= 0 ? magnitude << shift : (-shift >= 64 ? 0 : magnitude >> -shift);
        return fraction ? sign | uint32_t(fraction) : 0;
    }
    int shift         = length - 24;
    uint64_t mantissa = shift >= 0 ? magnitude >> shift : magnitude << -shift;
    return sign | uint32_t(exponent + 127) << 23 | (uint32_t(mantissa) & 0x7fffffu);
}

__global__ void
rounding_kernel(const uint8_t *a,
                const uint8_t *b,
                const uint8_t *sfa,
                const uint8_t *sfb,
                const float *native_c,
                const int64_t *destinations,
                const int64_t *experts,
                const double *xstats,
                const double *wstats,
                float *output,
                uint8_t *flags,
                int rows,
                int n,
                int e,
                int live) {
    int lane    = threadIdx.x & 15;
    int64_t pos = int64_t(blockIdx.x) * 8 + threadIdx.x / 16;
    if (pos >= int64_t(live) * n) return;
    int r = int(pos / n), col = int(pos % n);
    int64_t expert_id = experts[r], destination = destinations[r];
    bool valid = expert_id >= 0 && expert_id < e && destination >= 0 && destination < rows;
    int expert = valid ? int(expert_id) : 0, source_row = valid ? int(destination) : 0;
    unsigned mask = 0xffffu << ((threadIdx.x & 31) >= 16 ? 16 : 0);
    float c       = 0.0f;
    int keep      = 0;
    if (lane == 0 && valid) {
        c     = native_c[int64_t(source_row) * n + col];
        valid = isfinite(c);
        if (valid) keep = certify(c, xstats + int64_t(r) * 5, wstats + (int64_t(expert) * n + col) * 5);
    }
    int status = __shfl_sync(mask, lane == 0 ? int(valid) : 0, 0, 16);
    keep       = __shfl_sync(mask, keep, 0, 16);
    if (!status) {
        if (lane == 0) {
            output[pos] = CUDART_NAN_F;
            flags[pos]  = 2;
        }
        return;
    }
    if (keep) {
        if (lane == 0) {
            output[pos] = c;
            flags[pos]  = 0;
        }
        return;
    }
    uint32_t accumulator = 0;
    for (int begin = 0; begin < 1024; begin += 16) {
        int ma, mb, ra, rb;
        decode_bf(reconstructed_word(a, sfa, source_row, begin + lane, rows, 0, true), ma, ra);
        decode_bf(reconstructed_word(b, sfb, col, begin + lane, n, expert, false), mb, rb);
        int64_t product = int64_t(ma) * mb;
        int raw_product = product ? ra + rb : -10000;
        for (int offset = 1; offset < 16; offset <<= 1)
            raw_product = max(raw_product, __shfl_xor_sync(mask, raw_product, offset, 16));
        int encoded = (accumulator >> 23) & 255;
        int64_t cm  = (accumulator & 0x7fffffu) + (encoded ? 0x800000u : 0);
        if (accumulator & 0x80000000u) cm = -cm;
        int rc        = max(encoded, 1) - 127;
        int dominant  = max(raw_product, cm ? rc : -10000);
        int quantum   = dominant == -10000 ? 0 : dominant - 25;
        int64_t units = align_integer(product, ra + rb - 14, quantum);
        for (int offset = 1; offset < 16; offset <<= 1)
            units += __shfl_xor_sync(mask, static_cast<long long>(units), offset, 16);
        units += align_integer(cm, rc - 23, quantum);
        accumulator = normalize_rtz(units, quantum);
    }
    if (lane == 0) {
        output[pos] = __uint_as_float(accumulator);
        flags[pos]  = 1;
    }
}

__device__ void
max_amax(float *address, float value) {
    auto p       = reinterpret_cast<unsigned *>(address);
    unsigned old = *p;
    for (;;) {
        unsigned assumed = old;
        float previous   = __uint_as_float(assumed);
        float updated    = isnan(previous) || isnan(value) ? CUDART_NAN_F : fmaxf(previous, value);
        old              = atomicCAS(p, assumed, __float_as_uint(updated));
        if (old == assumed) break;
    }
}

__global__ void
activation_kernel(const float *c, const int64_t *experts, __nv_bfloat16 *d, float *amax, int live, int n, int e) {
    int64_t pos = int64_t(blockIdx.x) * blockDim.x + threadIdx.x;
    int width   = n / 2;
    if (pos >= int64_t(live) * width) return;
    int row = int(pos / width), column = int(pos % width);
    if (experts[row] < 0 || experts[row] >= e) {
        d[pos] = __float2bfloat16_rn(CUDART_NAN_F);
        return;
    }
    int64_t gate_pos = int64_t(row) * n + (column / 32) * 64 + column % 32;
    float gate       = __bfloat162float(__float2bfloat16_rn(c[gate_pos]));
    float up         = __bfloat162float(__float2bfloat16_rn(c[gate_pos + 32]));
    // Same explicit numerical nodes as the fixed independent staged oracle.
    float activated       = gate / (1.0f + expf(-gate));
    activated             = __bfloat162float(__float2bfloat16_rn(activated));
    __nv_bfloat16 product = __float2bfloat16_rn(__fmul_rn(activated, up));
    d[pos]                = product;
    max_amax(amax + experts[row], fabsf(__bfloat162float(product)));
}

void
guarded_rounding_cuda(const at::Tensor &a,
                      const at::Tensor &b,
                      const at::Tensor &sfa,
                      const at::Tensor &sfb,
                      const at::Tensor &native_c,
                      const at::Tensor &destinations,
                      const at::Tensor &experts,
                      const at::Tensor &xstats,
                      const at::Tensor &wstats,
                      const at::Tensor &output,
                      const at::Tensor &flags,
                      const at::Tensor &d,
                      const at::Tensor &amax) {
    for (const auto &tensor :
         {a, b, sfa, sfb, native_c, destinations, experts, xstats, wstats, output, flags, d, amax}) {
        TORCH_CHECK(tensor.is_cuda() && tensor.is_contiguous() && tensor.device() == a.device(),
                    "all tensors must be contiguous on one CUDA device");
    }
    TORCH_CHECK(a.dim() == 2 && a.size(1) == 4096 && a.size(0) % 128 == 0,
                "A must be padded K4096 four-component recipe");
    TORCH_CHECK(b.dim() == 3 && b.size(0) > 0 && b.size(1) > 0 && b.size(1) % 128 == 0 && b.size(2) == 4096,
                "B recipe shape invalid");
    TORCH_CHECK(a.scalar_type() == at::ScalarType::Float8_e4m3fn && b.scalar_type() == a.scalar_type(),
                "A/B must be E4M3");
    TORCH_CHECK(sfa.scalar_type() == at::ScalarType::Float8_e8m0fnu && sfb.scalar_type() == sfa.scalar_type(),
                "scales must be UE8M0");
    int rows = int(a.size(0)), n = int(b.size(1)), e = int(b.size(0)), live = int(destinations.numel());
    TORCH_CHECK(sfa.numel() == int64_t(rows) * 128 && sfb.numel() == int64_t(e) * n * 128,
                "scale storage size invalid");
    TORCH_CHECK(native_c.scalar_type() == at::kFloat && native_c.sizes() == at::IntArrayRef({rows, n}),
                "native C must be MxN FP32");
    TORCH_CHECK(destinations.scalar_type() == at::kLong && destinations.dim() == 1 &&
                    experts.scalar_type() == at::kLong && experts.sizes() == destinations.sizes(),
                "routes must be matching int64 vectors");
    TORCH_CHECK(xstats.scalar_type() == at::kDouble && xstats.sizes() == at::IntArrayRef({live, 5}) &&
                    wstats.scalar_type() == at::kDouble && wstats.sizes() == at::IntArrayRef({e, n, 5}),
                "input-only bound summaries invalid");
    TORCH_CHECK(output.scalar_type() == at::kFloat && output.sizes() == at::IntArrayRef({live, n}),
                "output C storage invalid");
    TORCH_CHECK(flags.scalar_type() == at::kByte && flags.sizes() == output.sizes(), "flag storage invalid");
    TORCH_CHECK(d.scalar_type() == at::kBFloat16 && d.sizes() == at::IntArrayRef({live, n / 2}), "D storage invalid");
    TORCH_CHECK(amax.scalar_type() == at::kFloat && amax.sizes() == at::IntArrayRef({e}), "AMAX storage invalid");
    c10::cuda::CUDAGuard guard(a.device());
    auto stream = c10::cuda::getCurrentCUDAStream(a.get_device()).stream();
    if (!live) return;
    rounding_kernel<<<(int64_t(live) * n + 7) / 8, 128, 0, stream>>>(reinterpret_cast<const uint8_t *>(a.data_ptr()),
                                                                     reinterpret_cast<const uint8_t *>(b.data_ptr()),
                                                                     reinterpret_cast<const uint8_t *>(sfa.data_ptr()),
                                                                     reinterpret_cast<const uint8_t *>(sfb.data_ptr()),
                                                                     native_c.data_ptr<float>(),
                                                                     destinations.data_ptr<int64_t>(),
                                                                     experts.data_ptr<int64_t>(),
                                                                     xstats.data_ptr<double>(),
                                                                     wstats.data_ptr<double>(),
                                                                     output.data_ptr<float>(),
                                                                     flags.data_ptr<uint8_t>(),
                                                                     rows,
                                                                     n,
                                                                     e,
                                                                     live);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
    activation_kernel<<<(int64_t(live) * (n / 2) + 255) / 256, 256, 0, stream>>>(
        output.data_ptr<float>(),
        experts.data_ptr<int64_t>(),
        reinterpret_cast<__nv_bfloat16 *>(d.data_ptr()),
        amax.data_ptr<float>(),
        live,
        n,
        e);
    C10_CUDA_KERNEL_LAUNCH_CHECK();
}
