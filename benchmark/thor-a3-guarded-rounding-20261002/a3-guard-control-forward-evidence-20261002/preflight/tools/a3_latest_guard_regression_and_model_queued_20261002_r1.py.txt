"""Fresh bed94f7e-based build, 31 production checks, native control boundaries/model.

Shared lock is acquired before any GPU import; prior source/binding stays
frozen. No other job interruption, service, timer, clock or power change.
"""

import argparse
import fcntl
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import sys
import time

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
HEAD = "5f1b6e17bbd741e3bbc2210a11434cb42bad3751"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    tools = Path(__file__).parent
    paths = [
        Path(__file__),
        tools / "run_a3_latest_bed_regression_locked_20261002_r1.py",
        tools / "a3_guarded_rounding_boundary_queued_20261002_r2.py",
        tools / "a3_guarded_rounding_full_model_20261002_r2.py",
        tools / "a3_repair_import_probe_20261001.py",
        tools / "a3_repair_matrix_select_20261001.py",
    ]
    before = {str(path): sha(path) for path in paths}
    native = json.loads((ROOT / "results/a3-latest-bed-native-provenance-20261002.json").read_text())
    assert native["native_build"] == "FRESH_RC0" and native["head"] == HEAD and native["base"] == "bed94f7e8362d3d0714fe40ab5aeb099c323949b"
    assert native["build_run"]["returncode"] == 0 and native["build_run"]["status"] == "EXITED"
    assert sha(native["binding"]["path"]) == native["binding"]["sha256"]
    args.expected_source = native["source"]
    args.expected_head = HEAD
    args.expected_binding_sha = native["binding"]["sha256"]
    (out / "queue_frozen_sources.json").write_text(json.dumps({"files": before, "native": native}, indent=2) + "\n")
    with open("/tmp/codex-thor-perf.lock", "a") as lock:
        start = time.monotonic()
        (out / "admission_wait.json").write_text(json.dumps({"stage": "WAITING_FOR_SHARED_LOCK", "pid": os.getpid(), "torch_imported": False}) + "\n")
        while True:
            try:
                fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                break
            except BlockingIOError:
                if time.monotonic() - start > 600:
                    (out / "admission.json").write_text(json.dumps({"stage": "NOT_RUN_SHARED_LOCK_BUSY", "torch_imported": False}) + "\n")
                    return 75
                time.sleep(1)
        assert "torch" not in sys.modules and {str(path): sha(path) for path in paths} == before
        assert all(sha(Path(args.expected_source) / item["path"]) == item["sha256"] for item in native["native_input_digests"])
        (out / "admission.json").write_text(
            json.dumps({"stage": "LOCK_ADMITTED", "pid": os.getpid(), "torch_imported": False, "wait_seconds": time.monotonic() - start}) + "\n"
        )
        os.environ["A3_LATEST_GUARD_LOCK_ADMITTED"] = "1"
        regression = load("a3_latest_bed_regression", paths[1])
        rc = regression.execute(out / "regression")
        if rc:
            return rc
        boundary = out / "boundaries"
        boundary.mkdir()
        b = load("a3_latest_native_guard_boundaries", paths[2])
        b.execute(args, boundary)
        model = out / "model"
        model.mkdir()
        g = load("a3_latest_native_guard_model", paths[3])
        g.execute(args, model, boundary)
        assert {str(path): sha(path) for path in paths} == before
        result = {
            "stage": "LATEST_REGRESSION_AND_NATIVE_ROUNDING_CONTROL_TERMINAL",
            "source_head": HEAD,
            "base": native["base"],
            "binding": native["binding"],
            "regression_result_sha256": sha(out / "regression/result.json"),
            "boundaries_result_sha256": sha(boundary / "result.json"),
            "model_result_sha256": sha(model / "result.json"),
            "frozen_sources": before,
            "model_stage": json.loads((model / "result.json").read_text())["stage"],
            "limits": "Costly precision control only. No generation/KV/formal performance or production precision acceptance.",
        }
        (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
        print(json.dumps(result), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
