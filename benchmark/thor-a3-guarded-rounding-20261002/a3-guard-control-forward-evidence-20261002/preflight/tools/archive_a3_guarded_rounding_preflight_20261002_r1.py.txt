"""Archive terminal preflight/CPU/admission evidence only; exclude PT/cache/binaries."""

import hashlib
import json
from pathlib import Path
import shutil
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
NAME = "a3-guarded-rounding-preflight-evidence-20261002"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    destination = ROOT / "results/a3-guarded-rounding-preflight-archive-20261002-r1"
    destination.mkdir(exist_ok=False)
    tree = destination / NAME
    tree.mkdir()
    directories = [
        "a3-certified-bf16-guard-cpu-20261001-r1",
        "a3-certified-guard-independent-cpu-audit-20261002-r1",
        "a3-guarded-rounding-build-cpu-20261001-r1",
        "a3-guarded-rounding-build-cpu-20261002-r2",
        "a3-guarded-rounding-build-cpu-20261002-r3",
        "a3-guarded-rounding-boundaries-20261002-r1",
        "a3-guarded-rounding-boundary-and-model-20261002-r1",
        "a3-latest-guard-regression-model-20261002-r1",
        "a3-postfixed-bound-cpu-20261002-r1",
    ]
    wanted, runs = [], []
    for name in directories:
        folder = ROOT / "results" / name
        launch = folder.with_name(name + "-launch")
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] in (0, 1, 75) and run["signal"] is None
        runs.append({"name": name, "run": run})
        wanted.extend(p for p in folder.glob("*.json") if p.is_file())
        wanted.extend(p for p in launch.iterdir() if p.is_file() and p.suffix in (".json", ".log"))
    for name in ("prepare-build-a3-latest-bed-20261002-r1-launch", "build-a3-latest-bed-20261002-r1"):
        folder = ROOT / "results" / name
        run = json.loads((folder / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] == 0
        runs.append({"name": name, "run": run})
        wanted.extend(p for p in folder.iterdir() if p.is_file() and p.suffix in (".json", ".log"))
    wanted.extend(
        ROOT / "results" / name
        for name in ("a3-latest-bed-prebuild-20261002.json", "a3-latest-bed-native-provenance-20261002.json", "a3-latest-e2-native-provenance-20261001.json")
    )
    sources = [
        "probe_a3_certified_bf16_guard_cpu_20261001.py",
        "audit_a3_certified_guard_cpu_20261002_r1.py",
        "probe_a3_postfixed_bound_cpu_20261002_r1.py",
        "build_a3_guarded_rounding_cpu_20261001.py",
        "build_a3_guarded_rounding_cpu_20261002_r2.py",
        "build_a3_guarded_rounding_cpu_20261002_r3.py",
        "a3_guarded_rounding_boundary_queued_20261002_r1.py",
        "a3_guarded_rounding_boundary_queued_20261002_r2.py",
        "a3_guarded_rounding_full_model_20261002_r1.py",
        "a3_guarded_rounding_full_model_20261002_r2.py",
        "a3_guarded_rounding_boundary_and_model_queued_20261002_r1.py",
        "a3_latest_guard_regression_and_model_queued_20261002_r1.py",
        "run_a3_latest_bed_regression_locked_20261002_r1.py",
        "prepare_build_a3_latest_bed_20261002_r1.py",
        "a3_mxfp8_bf16_staged_full_model_worker_20261001.py",
        "a3_mxfp8_bf16_staged_native_kernel_20261001.py",
        "a3_all24_precision_probe_20261001.py",
        "a3_model_e2e_mergefix_20261001.py",
        "validate_a3_mxfp8_window_cpu_20261001.py",
        "diagnose_a3_fixed_window_cpu_20261001.py",
        "a3_repair_import_probe_20261001.py",
        "a3_repair_matrix_select_20261001.py",
        Path(__file__).name,
    ]
    for stem in ("a3_guarded_rounding_native_20261001", "a3_guarded_rounding_native_20261002_r2", "a3_guarded_rounding_native_20261002_r3"):
        sources.extend(stem + extension for extension in (".cpp", ".cu", ".py"))
    wanted.extend(ROOT / "tools" / name for name in sources)
    manifest = []
    for path in sorted(set(wanted)):
        assert path.is_file() and path.suffix not in (".pt", ".so", ".o", ".plan", ".engine")
        relative = path.relative_to(ROOT)
        target = tree / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(path, target)
        assert sha(target) == sha(path)
        manifest.append({"path": str(relative), "size": path.stat().st_size, "sha256": sha(path)})
    status_path = tree / "TERMINAL_PREFLIGHT_RUNS.json"
    status_path.write_text(
        json.dumps(
            {
                "runs": runs,
                "native_gpu_work_admitted": False,
                "limits": "No actual native guard/31 latest checks/full24 model test ran; all three GPU admissions exited75 before Torch import. CPU compilation, CPU prediction and mathematical enclosure only.",
            },
            indent=2,
        )
        + "\n"
    )
    manifest.append({"path": status_path.name, "size": status_path.stat().st_size, "sha256": sha(status_path)})
    (tree / "MANIFEST.json").write_text(
        json.dumps({"files": manifest, "excluded": ["cache", "models", "engines", "PT", "native/JIT binaries", "Git bundle"]}, indent=2) + "\n"
    )
    tar = destination / (NAME + ".tar.gz")
    with tarfile.open(tar, "w:gz") as archive:
        archive.add(tree, arcname=NAME)
    result = {
        "path": str(tar),
        "sha256": sha(tar),
        "size": tar.stat().st_size,
        "manifest_entries": len(manifest),
        "file_count": len(manifest) + 1,
        "runs": len(runs),
    }
    (destination / "archive.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps(result), flush=True)


if __name__ == "__main__":
    main()
