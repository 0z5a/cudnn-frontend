"""Pinned-ID full Granite greedy/KV correctness, then frozen paired timing.

Actual MXFP8 main plus encoded-input integer repair, all24 layers. The native
expert forward is byte-identical to the qualified g4 forward. Diagnostic flag
reductions remain included in this control's measured cost. No BF16 gate/up
TC/Torch fallback, source/reference substitution or hardware state change.
New stock generation is a fresh paired reference, never a replacement for the
original frozen full-forward baseline. Stock teacher forcing uses the exact
oracle continuation so the original 5% gate compares identical contexts.
"""

import importlib.util
import json
from pathlib import Path
import statistics
import subprocess
import sys
import time


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def execute(args, out, boundary):
    runtime_path = Path(__file__).with_name("a3_guarded_rounding_model_runtime_20261002_r1.py")
    qualified_path = runtime_path.with_name("a3_guarded_rounding_full_model_20261002_r4.py")
    forward = lambda path: path.read_text().split("    def forward(", 1)[1].split("\n    adapters = []", 1)[0]
    assert forward(runtime_path) == forward(qualified_path)
    g = load("a3_guard_pinned_generation_runtime", runtime_path)
    r = g.prepare(args, out, boundary)
    m, t, h = r.m, r.t, r.h
    tokenizer = m.AutoTokenizer.from_pretrained(m.MODEL, local_files_only=True, trust_remote_code=False)
    paths = [Path(__file__), runtime_path, qualified_path, Path(r.q.__file__), Path(r.q.PROBE), Path(r.q.p.BASE), r.q.prototype_path, Path(h.__file__)]
    frozen = {str(path): m.sha_file(path) for path in paths}
    (out / "frozen_inputs.json").write_text(json.dumps({"files": frozen, "native_forward_byte_identical_to_g4": True}, indent=2) + "\n")
    saved, quality, timed, warmups = {}, [], [], []
    cases = [
        ("math73", r.prompts[0]),
        ("sky70", r.prompts[1]),
        ("passcode83", r.prompts[2]),
        ("prefix256", r.workloads["256"]),
        ("prefix1024", r.workloads["1024"]),
    ]
    start_main, start_repair, start_kernels = m.NATIVE_CALLS, h.POSTPROCESS_CALLS, h.POSTPROCESS_KERNEL_LAUNCHES

    def equal(a, b):
        return t.equal(a.contiguous().flatten().view(t.uint8), b.contiguous().flatten().view(t.uint8))

    def text(ids):
        tokens = ids[0].tolist()
        if tokenizer.eos_token_id in tokens:
            tokens = tokens[: tokens.index(tokenizer.eos_token_id)]
        return tokenizer.decode(tokens, skip_special_tokens=True).strip()

    def request(ids, mode, forced=None):
        m.attach(r.adapters, mode)
        for adapter in r.adapters:
            adapter.flag_counts.clear()
        bm, br, bk = m.NATIVE_CALLS, h.POSTPROCESS_CALLS, h.POSTPROCESS_KERNEL_LAUNCHES
        ids = ids.cuda()
        forced = forced.cuda() if forced is not None else None
        t.cuda.synchronize()
        begin = time.perf_counter_ns()
        result = r.model(input_ids=ids, use_cache=True, logits_to_keep=1)
        t.cuda.synchronize()
        prefill_done = time.perf_counter_ns()
        next_id = result.logits[:, -1].argmax(-1, keepdim=True)
        generated, logits = [next_id], [result.logits[:, -1].detach()]
        cache = result.past_key_values
        cache_type = type(cache).__name__
        for step in range(31):
            feed = next_id if forced is None else forced[:, step : step + 1]
            previous_cache = cache
            result = r.model(input_ids=feed, past_key_values=cache, use_cache=True, logits_to_keep=1)
            next_id = result.logits[:, -1].argmax(-1, keepdim=True)
            generated.append(next_id)
            logits.append(result.logits[:, -1].detach())
            cache = result.past_key_values
            assert cache is previous_cache, "KV cache was replaced instead of reused"
        t.cuda.synchronize()
        end = time.perf_counter_ns()
        assert cache.get_seq_length() == ids.shape[-1] + 31
        counts = (m.NATIVE_CALLS - bm, h.POSTPROCESS_CALLS - br, h.POSTPROCESS_KERNEL_LAUNCHES - bk)
        assert counts == ((768, 768, 1536) if mode == "native" else (0, 0, 0))
        flags = t.stack([t.stack(adapter.flag_counts) for adapter in r.adapters]).cpu() if mode == "native" else None
        if flags is not None:
            assert flags.shape == (24, 32, 3) and int(flags[:, :, 2].sum()) == 0
        return {
            "ids": t.cat(generated, dim=-1).cpu(),
            "logits": t.stack(logits, dim=1).cpu(),
            "flags": flags,
            "timing": {"prefill_ms": (prefill_done - begin) / 1e6, "decode_ms": (end - prefill_done) / 1e6, "request_ms": (end - begin) / 1e6},
            "native_main_calls": counts[0],
            "native_repair_calls": counts[1],
            "native_postprocess_kernel_launches": counts[2],
            "cache_type": cache_type,
            "cache_final_tokens": cache.get_seq_length(),
            "cache_object_reused": True,
            "forwards": 32,
        }

    def flush():
        t.save(saved, out / "partial_outputs.pt")
        (out / "partial.json").write_text(json.dumps({"quality": quality, "warmups": warmups, "timed": timed}, indent=2) + "\n")

    quality_outputs = {}
    with t.inference_mode():
        for name, ids in cases:
            saved[name + "/input_ids"] = ids
            outputs = {}
            for mode in ("reference", "native", "stock"):
                value = request(ids, mode)
                outputs[mode] = value
                saved[name + "/" + mode + "/ids"] = value["ids"]
                saved[name + "/" + mode + "/logits"] = value["logits"]
                if value["flags"] is not None:
                    saved[name + "/native/flags"] = value["flags"]
                print(
                    json.dumps(
                        {
                            "stage": "GENERATION_ARM_COMPLETE",
                            "case": name,
                            "mode": mode,
                            "timing": value["timing"],
                            "native_main_calls": value["native_main_calls"],
                        }
                    ),
                    flush=True,
                )
                flush()
            teacher = request(ids, "stock", forced=outputs["reference"]["ids"])
            saved[name + "/stock_teacher/logits"] = teacher["logits"]
            saved[name + "/stock_teacher/ids"] = teacher["ids"]
            native = outputs["native"]
            oracle = outputs["reference"]
            oracle_error, stock_error = m.difference(native["logits"], oracle["logits"]), m.difference(native["logits"], teacher["logits"])
            expected = "4" if name == "math73" else "cobalt" if name in ("passcode83", "prefix256", "prefix1024") else None
            decoded = {mode: text(value["ids"]) for mode, value in outputs.items()}
            row = {
                "case": name,
                "input_tokens": ids.shape[-1],
                "new_tokens": 32,
                "ids": m.tensor_info(ids),
                "native_oracle": oracle_error,
                "native_original_bf16_same_context": stock_error,
                "oracle_gate_pass": equal(native["ids"], oracle["ids"]) and oracle_error["relative_l2"] <= 0.01,
                "bf16_gate_pass": equal(native["ids"], oracle["ids"]) and stock_error["relative_l2"] <= 0.05,
                "same_context_for_numeric_gates": equal(native["ids"], oracle["ids"]),
                "native_oracle_logits_byte_equal": equal(native["logits"], oracle["logits"]),
                "native_oracle_ids_byte_equal": equal(native["ids"], oracle["ids"]),
                "native_stock_greedy_ids_byte_equal": equal(native["ids"], outputs["stock"]["ids"]),
                "texts": decoded,
                "expected": expected,
                "answer_pass": bool(decoded["native"]) and (expected is None or expected in decoded["native"].lower()),
                "flag_totals": native["flags"].sum((0, 1)).tolist(),
                "native_main_calls": native["native_main_calls"],
                "native_repair_calls": native["native_repair_calls"],
                "native_postprocess_kernel_launches": native["native_postprocess_kernel_launches"],
                "cache_type": native["cache_type"],
                "cache_final_tokens": native["cache_final_tokens"],
                "cache_object_reused": True,
            }
            row["pass"] = all(row[k] for k in ("oracle_gate_pass", "bf16_gate_pass", "native_oracle_ids_byte_equal", "answer_pass"))
            quality.append(row)
            quality_outputs[name] = outputs
            flush()
            print(json.dumps({"stage": "GENERATION_QUALITY_CHECKED", **row}), flush=True)
        quality_pass = all(row["pass"] for row in quality)
        if quality_pass:
            for name, ids in cases[3:]:
                for mode in ("stock", "native"):
                    value = request(ids, mode)
                    known = quality_outputs[name][mode]
                    assert equal(value["ids"], known["ids"]) and equal(value["logits"], known["logits"])
                    warmups.append({"case": name, "mode": mode, "timing": value["timing"], "outputs_byte_equal_to_quality": True})
                for repeat, order in enumerate((("stock", "native", "native", "stock"), ("native", "stock", "stock", "native"))):
                    for position, mode in enumerate(order):
                        value = request(ids, mode)
                        known = quality_outputs[name][mode]
                        assert equal(value["ids"], known["ids"]) and equal(value["logits"], known["logits"])
                        timed.append(
                            {
                                "case": name,
                                "input_tokens": ids.shape[-1],
                                "new_tokens": 32,
                                "repeat": repeat,
                                "position": position,
                                "mode": mode,
                                **value["timing"],
                                "outputs_byte_equal_to_quality": True,
                                "native_main_calls": value["native_main_calls"],
                                "native_repair_calls": value["native_repair_calls"],
                                "cache_final_tokens": value["cache_final_tokens"],
                            }
                        )
                        flush()
                        print(json.dumps({"stage": "PAIRED_TIMING_OBSERVATION", **timed[-1]}), flush=True)
        m.attach(r.adapters, "stock")
    comparisons = []
    for name, ids in cases[3:]:
        arms = {mode: [row for row in timed if row["case"] == name and row["mode"] == mode] for mode in ("stock", "native")}
        if timed:
            assert len(arms["stock"]) == len(arms["native"]) == 4
            medians = {
                mode: {key: statistics.median(row[key] for row in rows) for key in ("prefill_ms", "decode_ms", "request_ms")} for mode, rows in arms.items()
            }
            comparisons.append(
                {
                    "case": name,
                    "median_ms": medians,
                    "stock_over_native_request_ratio": medians["stock"]["request_ms"] / medians["native"]["request_ms"],
                    "native_faster": medians["native"]["request_ms"] < medians["stock"]["request_ms"],
                }
            )
    assert {str(path): m.sha_file(path) for path in paths} == frozen
    assert all(m.sha_file(path) == digest for path, digest in r.build["sources"].items())
    assert m.sha_file(r.build["binary"]["path"]) == r.build["binary"]["sha256"]
    assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
    assert not subprocess.check_output(["git", "-C", args.expected_source, "status", "--porcelain", "--untracked-files=no"], text=True).strip()
    t.save(saved, out / "outputs.pt")
    result = {
        "stage": "PINNED_GENERATION_KV_AND_PAIRED_TIMING_COMPLETE" if quality_pass else "PINNED_GENERATION_KV_QUALITY_FAILED_PERFORMANCE_NOT_RUN",
        "scope": __doc__,
        "head": args.expected_head,
        "source": args.expected_source,
        "native_forward_byte_identical_to_g4": True,
        "quality_all_five_cases_pass": quality_pass,
        "quality": quality,
        "warmups": warmups,
        "timed": timed,
        "comparisons": comparisons,
        "native_main_calls": m.NATIVE_CALLS - start_main,
        "native_repair_calls": h.POSTPROCESS_CALLS - start_repair,
        "native_postprocess_kernel_launches": h.POSTPROCESS_KERNEL_LAUNCHES - start_kernels,
        "model_parameters": 1334628352,
        "layers": 24,
        "experts": 32,
        "topk": 8,
        "hidden": 1024,
        "intermediate": 512,
        "gemm_k": 4096,
        "files": frozen,
        "outputs_sha256": m.sha_file(out / "outputs.pt"),
        "original_input_tokens_json_sha256": m.sha_file(r.input_file),
        "original_forward_baseline_replaced": False,
        "oracle_replaced": False,
        "native_compute": "MXFP8_TC_then_encoded_only_integer_CUDA_repair",
        "native_flag_instrumentation_included_in_timing": True,
        "imports": {
            name: {"path": imported.__file__, "sha256": m.sha_file(imported.__file__)}
            for name, imported in (("cudnn", m.cudnn), ("api", m.api), ("binding", m._compiled_module))
        },
        "service_final": "inactive",
        "production_feature_accepted": False,
        "limits": "Five actual full-model cached 32-token generations. Original 1%/5% gates, stock teacher forcing at the oracle's exact context. Timings only after all numeric, token and answer checks; ABBA/BAAB four observations per arm at 256/1024, diagnostics included. No model/token/capacity reduction, no production code edit, no claimed throughput benefit from control alone.",
    }
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({"stage": result["stage"], "quality_all_five_cases_pass": quality_pass, "comparisons": comparisons}), flush=True)
