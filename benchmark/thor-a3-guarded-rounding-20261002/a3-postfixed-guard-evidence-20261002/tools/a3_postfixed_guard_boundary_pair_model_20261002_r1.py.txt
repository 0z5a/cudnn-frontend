"""New postfixed CUDA enclosure: boundaries, exact pair and full24 feedback.

The earlier CPU enclosure is looser and repairs more outputs. Test whether
avoiding192 sequential FP64 bound updates lowers actual repair-wrapper cost.
This is a separate candidate, not a promotion or a claim of model speedup.
"""

import importlib.util
import json
import math
from pathlib import Path
import statistics
import sys


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def execute(args, out):
    assert "torch" not in sys.modules
    tools = Path(__file__).parent
    boundary = out / "boundaries"
    boundary.mkdir()
    b = load("a3_postfixed_guard_actual_boundaries", tools / "a3_guarded_rounding_boundary_queued_20261002_r4.py")
    b.execute(args, boundary)
    q, h = sys.modules["a3_guard_boundary_quant"], sys.modules["a3_guard_boundary_cuda"]
    m, t = q.m, q.t
    build = json.loads((m.ROOT / "results/a3-guarded-rounding-build-cpu-20261002-r4/result.json").read_text())
    module = h.load_native(build["binary"]["path"], build["binary"]["sha256"])
    oldh = load("a3_guard_old_serial_control", tools / "a3_guarded_rounding_native_20261002_r3.py")
    oldbuild = json.loads((m.ROOT / "results/a3-guarded-rounding-build-cpu-20261002-r3/result.json").read_text())
    oldmodule = oldh.load_native(oldbuild["binary"]["path"], oldbuild["binary"]["sha256"])
    frozen = {
        str(path): m.sha_file(path) for path in [Path(__file__), Path(b.__file__), Path(oldh.__file__), tools / "a3_guarded_rounding_full_model_20261002_r5.py"]
    }
    saved, synthetic, paired = {}, [], []

    def up(value):
        return math.nextafter(value, math.inf) if value else 0.0

    def exponent(value):
        return math.frexp(value)[1] - 1

    def loss(s, error, raw, group):
        upper = up(s + error)
        alignment = math.ldexp(float(group + 1), max(exponent(upper), raw) - 25)
        intermediate = up(upper + alignment)
        ulp = math.ldexp(1.0, max(exponent(intermediate) - 23, -149))
        return up(alignment + ulp)

    def serial(s, raw, group, updates):
        if not s:
            return 0.0
        error = 0.0
        for _ in range(updates):
            error = up(error + loss(s, error, raw, group))
        return error

    def postfixed(s, raw, group, updates):
        if not s:
            return 0.0
        error = up(4.0 * updates * loss(s, 0.0, raw, group))
        for _ in range(8):
            rhs = up(updates * loss(s, error, raw, group))
            if error >= up(2.0 * rhs):
                return error
            error = up(error * 2.0)
        return math.inf

    bases = [
        math.ldexp(1.0, exponent) for exponent in [-298, -252, -149, -126, -64, -32, -16, -8, -4, -2, -1, 0, 1, 2, 4, 8, 16, 32, 64, 96, 126, 127, 192, 254]
    ]
    values = [0.0] + bases + [math.nextafter(value, 0.0) for value in bases] + [up(value) for value in bases]
    for group, updates in ((32, 128), (16, 64)):
        for delta in (-12, 0, 6, 20):
            raw = [exponent(value or 1.0) + delta for value in values]
            s_gpu, raw_gpu = t.tensor(values, dtype=t.float64, device="cuda"), t.tensor(raw, dtype=t.int64, device="cuda")
            actual = t.empty_like(s_gpu)
            module.bounds(s_gpu, raw_gpu, actual, group, updates)
            actual = actual.cpu()
            expected = t.tensor([postfixed(s, p, group, updates) for s, p in zip(values, raw)], dtype=t.float64)
            original = t.tensor([serial(s, p, group, updates) for s, p in zip(values, raw)], dtype=t.float64)
            assert t.equal(actual.view(t.int64), expected.view(t.int64)) and (actual >= original).all()
            key = f"bounds/{group}/{delta}"
            saved.update(
                {key + "/actual": actual, key + "/expected": expected, key + "/serial": original, key + "/s": s_gpu.cpu(), key + "/raw": raw_gpu.cpu()}
            )
            synthetic.append(
                {"group": group, "updates": updates, "raw_delta": delta, "count": len(values), "native_cpu_bit_equal": True, "serial_enclosed": True}
            )
    # BF16/F32 share the exponent field, including BF16 subnormals. Verify
    # the original stats expression across every finite BF16 bit pattern.
    words = t.arange(65536, dtype=t.int32).to(t.int16)
    bf = words.view(t.bfloat16)
    valid = t.isfinite(bf)
    oldraw = ((bf.float().view(t.int32) >> 23) & 255).clamp(min=1) - 127
    braw = ((words.to(t.int32) >> 7) & 255).clamp(min=1) - 127
    assert t.equal(oldraw[valid], braw[valid])
    captured = t.load(boundary / "outputs.pt", map_location="cpu", weights_only=True)
    rows = json.loads((boundary / "result.json").read_text())["rows"]
    with t.inference_mode():
        for row in rows:
            name, v = row["name"], captured[row["name"]]
            inputs = (
                v["a_codes"].cuda().view(t.float8_e4m3fn),
                v["b_codes"].cuda().view(t.float8_e4m3fn),
                v["sfa"].cuda().view(t.float8_e8m0fnu),
                v["sfb"].cuda().view(t.float8_e8m0fnu),
                v["native_c"].cuda(),
                v["destinations"].cuda(),
                v["experts"].cuda(),
                v["xstats"].cuda(),
                v["wstats"].cuda(),
            )
            for mode in ("serial", "postfixed"):
                worker, native = (oldh, oldmodule) if mode == "serial" else (h, module)
                worker.run(native, *inputs)
            t.cuda.synchronize()
            for repeat in range(5):
                order = ("serial", "postfixed", "postfixed", "serial") if repeat % 2 == 0 else ("postfixed", "serial", "serial", "postfixed")
                for position, mode in enumerate(order):
                    worker, native = (oldh, oldmodule) if mode == "serial" else (h, module)
                    before, after = t.cuda.Event(enable_timing=True), t.cuda.Event(enable_timing=True)
                    before.record()
                    value = worker.run(native, *inputs)
                    after.record()
                    after.synchronize()
                    elapsed = before.elapsed_time(after)
                    cpu = {key: tensor.cpu() for key, tensor in value.items()}
                    for key, ref in (("d", "reference_d"), ("amax", "reference_amax")):
                        assert t.equal(cpu[key].flatten().view(t.uint8), v[ref].flatten().view(t.uint8))
                    assert t.equal(cpu["c"].to(t.bfloat16).view(t.int16), v["reference_c_bf16"].view(t.int16)) and not (cpu["flags"] == 2).any()
                    saved.update({f"{name}/{repeat}/{position}/{key}": tensor for key, tensor in cpu.items()})
                    paired.append(
                        {
                            "case": name,
                            "repeat": repeat,
                            "position": position,
                            "mode": mode,
                            "gpu_wrapper_ms": elapsed,
                            "bf16_c_d_amax_reference_byte_equal": True,
                            "flags": [int((cpu["flags"] == i).sum()) for i in (0, 1, 2)],
                        }
                    )
            print(
                json.dumps(
                    {
                        "stage": "FROZEN_REPAIR_PAIR_COMPLETE",
                        "case": name,
                        "medians_ms": {
                            mode: statistics.median(item["gpu_wrapper_ms"] for item in paired if item["case"] == name and item["mode"] == mode)
                            for mode in ("serial", "postfixed")
                        },
                    }
                ),
                flush=True,
            )
    t.save(saved, out / "pair_outputs.pt")
    (out / "pair.json").write_text(
        json.dumps(
            {
                "synthetic": synthetic,
                "synthetic_count": sum(row["count"] for row in synthetic),
                "bf16_exponents_checked": int(valid.sum()),
                "paired": paired,
                "files": frozen,
                "outputs_sha256": m.sha_file(out / "pair_outputs.pt"),
                "old_build": oldbuild,
                "new_build": build,
            },
            indent=2,
        )
        + "\n"
    )
    model = out / "model"
    model.mkdir()
    g = load("a3_postfixed_guard_full24", tools / "a3_guarded_rounding_full_model_20261002_r5.py")
    g.execute(args, model, boundary)
    assert {path: m.sha_file(path) for path in frozen} == frozen
    result = {
        "stage": "POSTFIXED_BOUNDARY_PAIR_FULL24_TERMINAL",
        "head": args.expected_head,
        "boundary_result_sha256": m.sha_file(boundary / "result.json"),
        "model_stage": json.loads((model / "result.json").read_text())["stage"],
        "pair_result_sha256": m.sha_file(out / "pair.json"),
        "files": frozen,
        "generation_kv": "NOT_RUN_NEW_CANDIDATE",
        "paired_whole_model_performance": "NOT_RUN",
        "accepted_optimization": False,
        "limits": "Actual new conservative bound, native boundaries, 584 enclosure comparisons, 80 paired repair-wrapper observations and original3 full24 forwards. Wrapper timing is not whole-model throughput or completion; prior application passcode83 failure remains recorded.",
    }
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps(result), flush=True)
