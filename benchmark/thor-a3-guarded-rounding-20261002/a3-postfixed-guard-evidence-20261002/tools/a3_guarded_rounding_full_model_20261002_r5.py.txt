"""Full24 Granite feedback for the costly native integer rounding control.

Actual frozen MXFP8 Tensor Core C -> native encoded-input repair -> native
plain BF16 D is consumed by all24 layers. Repair accepts no source BF16
gate/up operand or oracle C. Its gate/up branch executes neither a Torch dot
nor a BF16 Tensor Core dot; original BF16 down projection stays unchanged.
This tests precision only. It is not a proposed efficient implementation.
"""

import json
from pathlib import Path
import subprocess
import sys
from types import MethodType


def execute(args, out, boundary):
    assert json.loads((boundary / "result.json").read_text())["stage"] == "NATIVE_ROUNDING_CONTROL_BOUNDARIES_COMPLETE_NOT_MODEL_OR_PERFORMANCE_SUCCESS"
    q = sys.modules["a3_guard_boundary_quant"]
    h = sys.modules["a3_guard_boundary_cuda"]
    m, t = q.m, q.t
    build_dir = m.ROOT / "results/a3-guarded-rounding-build-cpu-20261002-r4"
    build = json.loads((build_dir / "result.json").read_text())
    module = h.load_native(build["binary"]["path"], build["binary"]["sha256"])
    before_main, before_repair, before_kernels = m.NATIVE_CALLS, h.POSTPROCESS_CALLS, h.POSTPROCESS_KERNEL_LAUNCHES
    assert m.quantize_mxfp8 is q.quantize
    assert m.sha_file(m.MODEL / "model.safetensors") == m.WEIGHT_SHA
    source = Path(args.expected_source)
    assert subprocess.check_output(["git", "-C", str(source), "rev-parse", "HEAD"], text=True).strip() == args.expected_head
    baseline_dir = m.ROOT / "results/a3-model-repair-a3sfdlatest4-n256-20261001"
    original_baseline_sha = "a997dac0bbd8b137ecbf7ddbefc5f356c2633d75a9d3cde1d035335cb028440f"
    assert m.sha_file(baseline_dir / "outputs.pt") == original_baseline_sha
    baseline_meta = json.loads((baseline_dir / "result.json").read_text())
    assert baseline_meta["source_head"] == "7bfeda6bfeaa63a1e59fadc5d9fb48676f03f744"
    baseline = t.load(baseline_dir / "outputs.pt", map_location="cpu", weights_only=True)
    old_dir = m.ROOT / "results/a3-mxfp8-bf16-staged-full-model-20261001-r1"
    prior_oracle_sha = "96dcdce434fd9efce3bc3045097725eb1024cdfeb68a68cadbddc78306b3babe"
    assert m.sha_file(old_dir / "outputs.pt") == prior_oracle_sha
    old_meta = json.loads((old_dir / "result.json").read_text())
    assert old_meta["source_head"] == "741e10ffa5e93dbc0a8a41e06e66e0c81c1f85b1" and old_meta["experimental_native_kernel"]["sha256"] == m.sha_file(
        q.prototype_path
    )
    old = t.load(old_dir / "outputs.pt", map_location="cpu", weights_only=True)
    model, loading = m.AutoModelForCausalLM.from_pretrained(
        m.MODEL, dtype=t.bfloat16, attn_implementation="sdpa", trust_remote_code=False, local_files_only=True, output_loading_info=True
    )
    assert not any(loading.values()) and sum(v.numel() for v in model.parameters()) == 1334628352
    model = model.eval().cuda()
    experts = [expert for expert in model.modules() if isinstance(expert, m.GraniteMoeExperts)]
    assert len(experts) == 24
    # The pinned chat template inserts today's date. Recreating text changes
    # input IDs after midnight; use and hash-check the ORIGINAL baseline IDs.
    input_file = baseline_dir / "input-token-ids.json"
    original_ids = json.loads(input_file.read_text())
    prompts = [t.tensor(ids, dtype=t.int64) for ids in original_ids["prompts"]]
    workloads = {key: t.tensor(ids, dtype=t.int64) for key, ids in original_ids["workloads"].items()}
    assert all(m.tensor_info(ids) == expected for ids, expected in zip(prompts, baseline_meta["inputs"]["prompts"], strict=True))
    assert all(m.tensor_info(ids) == baseline_meta["inputs"]["workloads"][key] for key, ids in workloads.items())
    cases = [prompts[0], prompts[1], workloads["256"]]
    assert [ids.shape[-1] for ids in cases] == [73, 70, 256]

    def forward(self, hidden, index, probability):
        self.calls += 1
        if self.mode == "reference":
            return self.reference_staged(hidden, index, probability)
        assert self.mode == "native" and m.quantize_mxfp8 is q.quantize
        x, ends, destinations, order, token_ids, sizes, aligned = self.routing(hidden, index)
        self.last_counts = sizes
        a, sfa, sa, unused_effective = q.expanded_quantize(x, "A")
        # Bound summaries are computed solely from the already encoded recipe.
        xs = h.recipe_stats(a, sa, "A", destinations)
        expert_ids = index.flatten()[order]
        stream = t.cuda.current_stream()
        actual = m.api.grouped_gemm_swiglu_wrapper_sm100(
            a_tensor=a,
            b_tensor=self.b,
            sfa_tensor=sfa,
            sfb_tensor=self.sfb,
            padded_offsets=ends,
            alpha_tensor=self.alpha,
            norm_const_tensor=self.norm,
            prob_tensor=None,
            acc_dtype=t.float32,
            c_dtype=t.float32,
            d_dtype=t.bfloat16,
            mma_tiler_mn=self.tile,
            cluster_shape_mn=(2, 1),
            sf_vec_size=32,
            m_aligned=256,
            current_stream=m.cuda.CUstream(stream.cuda_stream),
        )
        corrected = h.run(module, a, self.b, sfa, self.sfb, actual["c_tensor"], destinations, expert_ids, xs, self.wstats, stream=stream)
        self.flag_counts.append(t.stack([(corrected["flags"] == i).sum() for i in (0, 1, 2)]))
        # The actually emitted native plain BF16 D is the model input here.
        live = corrected["d"]
        routes = t.empty((order.numel(), self.h), dtype=t.bfloat16, device=x.device)
        weights = probability.flatten()[order]
        start = 0
        for expert_id, count in enumerate(sizes):
            if count:
                y = m.F.linear(live[start : start + count], self.expert.down_proj[expert_id])
                routes[start : start + count] = (y * weights[start : start + count, None]).to(t.bfloat16)
            start += count
        unsorted = t.empty_like(routes)
        unsorted[order] = routes
        return unsorted.reshape(hidden.shape[0], index.shape[-1], self.h).sum(1)

    adapters = []
    with t.inference_mode():
        for layer, expert in enumerate(experts):
            adapter = q.make_adapter(m.A3ExpertAdapter, expert, layer, bf16_stage=True)
            adapter.wstats = h.recipe_stats(adapter.b, adapter.bscale, "B")
            adapter.flag_counts = []
            adapter.forward = MethodType(forward, adapter)
            adapters.append(adapter)
    rows, saved = [], {}
    frozen_paths = [Path(__file__), Path(q.__file__), Path(q.PROBE), Path(q.p.BASE), q.prototype_path, Path(h.__file__)]
    frozen = {str(path): m.sha_file(path) for path in frozen_paths}
    (out / "frozen_inputs.json").write_text(
        json.dumps({"files": frozen, "build": build, "boundary_result_sha256": m.sha_file(boundary / "result.json")}, indent=2) + "\n"
    )
    with t.inference_mode():
        for case, ids in enumerate(cases):
            saved[f"{case}/input_ids"] = ids.cpu()
            m.attach(adapters, "reference")
            before = m.NATIVE_CALLS
            oracle = model(input_ids=ids.cuda(), use_cache=False).logits.cpu()
            assert m.NATIVE_CALLS == before
            saved[f"{case}/reference"] = oracle
            t.save(saved, out / "partial_outputs.pt")
            prior_oracle = old[f"k32_mxfp8_dot_bf16_stage/{case}/reference"]
            assert t.equal(
                oracle.flatten().view(t.uint8), prior_oracle.flatten().view(t.uint8)
            ), "Declared oracle changed from the frozen failed-control experiment"
            saved[f"{case}/reference"] = oracle
            for adapter in adapters:
                adapter.flag_counts.clear()
            m.attach(adapters, "native")
            bm, br, bk = m.NATIVE_CALLS, h.POSTPROCESS_CALLS, h.POSTPROCESS_KERNEL_LAUNCHES
            native = model(input_ids=ids.cuda(), use_cache=False).logits.cpu()
            assert m.NATIVE_CALLS - bm == 24 and h.POSTPROCESS_CALLS - br == 24 and h.POSTPROCESS_KERNEL_LAUNCHES - bk == 48
            assert all(len(adapter.flag_counts) == 1 for adapter in adapters)
            flags = t.stack([adapter.flag_counts[0] for adapter in adapters]).cpu()
            saved[f"{case}/native"] = native
            saved[f"{case}/flag_counts"] = flags
            stock = baseline[f"forward_{case}_stock"]
            assert native.shape == oracle.shape == stock.shape == (1, (73, 70, 256)[case], 49155)
            oracle_error, bf16_error = m.difference(native, oracle), m.difference(native, stock)
            row = {
                "case": case,
                "input_tokens": ids.shape[-1],
                "mxfp8_layers": list(range(24)),
                "gemm_k": 4096,
                "model_parameters": 1334628352,
                "native_main_calls": 24,
                "native_repair_calls": 24,
                "native_postprocess_kernel_launches": 48,
                "native_oracle": oracle_error,
                "native_bf16": bf16_error,
                "oracle_gate_pass": oracle_error["relative_l2"] <= 0.01,
                "bf16_gate_pass": bf16_error["relative_l2"] <= 0.05,
                "native_oracle_byte_equal": t.equal(native.flatten().view(t.uint8), oracle.flatten().view(t.uint8)),
                "oracle_byte_equal_to_prior_failed_control": True,
                "oracle_bf16_compatibility": q.compatibility(oracle, stock),
                "flag_counts_per_layer": flags.tolist(),
                "ambiguous_fraction": (flags[:, 1].sum() / flags.sum()).item(),
            }
            rows.append(row)
            print(json.dumps(row), flush=True)
            (out / "partial.json").write_text(json.dumps({"rows": rows}, indent=2) + "\n")
            t.save(saved, out / "partial_outputs.pt")
            assert int(flags[:, 2].sum()) == 0, "Native control emitted an invalid result"
        m.attach(adapters, "stock")
    assert {str(path): m.sha_file(path) for path in frozen_paths} == frozen
    assert all(m.sha_file(path) == digest for path, digest in build["sources"].items())
    assert m.sha_file(build["binary"]["path"]) == build["binary"]["sha256"]
    assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
    assert not subprocess.check_output(["git", "-C", str(source), "status", "--porcelain", "--untracked-files=no"], text=True).strip()
    t.save(saved, out / "outputs.pt")
    passed = all(row["oracle_gate_pass"] and row["bf16_gate_pass"] for row in rows)
    result = {
        "stage": "ALL_THREE_FORWARD_GATES_PASS_COSTLY_PRECISION_CONTROL" if passed else "FULL24_NATIVE_ROUNDING_CONTROL_GATES_FAILED",
        "scope": __doc__,
        "rows": rows,
        "all_three_cases_both_gates": passed,
        "source": str(source),
        "source_head": args.expected_head,
        "checkpoint": baseline_meta["checkpoint"],
        "checkpoint_sha256": m.WEIGHT_SHA,
        "model_parameters": 1334628352,
        "experts": 32,
        "topk": 8,
        "hidden": 1024,
        "intermediate": 512,
        "vocabulary": 49155,
        "native_main_calls": m.NATIVE_CALLS - before_main,
        "native_repair_calls": h.POSTPROCESS_CALLS - before_repair,
        "native_postprocess_kernel_launches": h.POSTPROCESS_KERNEL_LAUNCHES - before_kernels,
        "native_plans": m.NATIVE_PLANS,
        "files": frozen,
        "native_build_result_sha256": m.sha_file(build_dir / "result.json"),
        "native_binary": build["binary"],
        "boundary_result_sha256": m.sha_file(boundary / "result.json"),
        "original_baseline_head": baseline_meta["source_head"],
        "original_baseline_outputs_sha256": original_baseline_sha,
        "original_baseline_replaced": False,
        "input_tokens": [m.tensor_info(ids) for ids in cases],
        "original_input_tokens_json_sha256": m.sha_file(input_file),
        "input_ids_regenerated_from_current_date": False,
        "prior_failed_control_outputs_sha256": prior_oracle_sha,
        "declared_oracle_replaced": False,
        "production_source_modified": False,
        "imports": {
            name: {"path": imported.__file__, "sha256": m.sha_file(imported.__file__)}
            for name, imported in (("cudnn", m.cudnn), ("api", m.api), ("binding", m._compiled_module))
        },
        "outputs_sha256": m.sha_file(out / "outputs.pt"),
        "service_final": "inactive",
        "formal_performance": "NOT_RUN",
        "generation_kv": "NOT_RUN",
        "limits": "Actual whole-model input forward only, all24 layers and full logits under original 1% oracle/5% original BF16 gates. Conservative guard may recompute nearly every C from encoded operands with integer ALU. This is a costly precision control, not an accepted efficient implementation, production feature, generation/KV or throughput result. Pure MXFP8 main retained; no native BF16 gate/up TC GEMM or Torch gate/up dot in native branch.",
    }
    assert result["native_main_calls"] == result["native_repair_calls"] == 72 and result["native_postprocess_kernel_launches"] == 144
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({key: result[key] for key in ("stage", "all_three_cases_both_gates", "native_main_calls", "native_repair_calls")}), flush=True)
