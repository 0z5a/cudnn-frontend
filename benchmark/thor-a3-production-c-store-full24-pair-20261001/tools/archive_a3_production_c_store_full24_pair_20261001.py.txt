"""Archive terminal C-store diagnostics, failed harnesses and verified private tests."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-production-c-store-full24-pair-evidence-20261001"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    runs = {"a3-production-c-store-full24-pair-20261001-r1": 0, "a3-production-c-store-full24-pair-cpu-audit-20261001-r1": 0}
    selected = {}
    for name, expected in runs.items():
        directory = ROOT / "results" / name
        launch = directory.with_name(name + "-launch")
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["signal"] is None and run["returncode"] == expected
        for base in (directory, launch):
            for path in sorted(base.rglob("*")):
                if path.is_file() and path.suffix in (".json", ".log", ".xml", ".txt"):
                    selected[str(path.relative_to(ROOT))] = path
    assert (
        json.loads((ROOT / "results/a3-production-c-store-full24-pair-cpu-audit-20261001-r1/result.json").read_text())["stage"]
        == "CPU_AUDITED_FULL24_PRODUCTION_C_STORE_BYTE_PARITY_QUALITY_STILL_FAILED"
    )
    for name in (
        "a3_production_c_store_full24_pair_queued_20261001.py",
        "audit_a3_production_c_store_full24_pair_20261001.py",
        "archive_a3_production_c_store_full24_pair_20261001.py",
        "a3_production_optional_c_store_native_kernel_20261001.py",
        "a3_model_e2e_mergefix_20261001.py",
        "audit_a3_precision_followup_r2_20261001.py",
    ):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    for name in (
        "a3-latest-e2-native-provenance-20261001.json",
        "a3-mxfp8-bf16-staged-full-model-20261001-r1/result.json",
        "a3-n128-mxfp8-staged-trace-cpu-audit-20261001-r1/result.json",
    ):
        path = ROOT / "results" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    source = ROOT / "a3sfdlateste2"
    for relative in (
        "python/cudnn/gemm/cutedsl/grouped/swiglu/grouped_gemm_swiglu_quant.py",
        "python/cudnn/gemm/cutedsl/grouped/swiglu/api.py",
        "test/python/gemm/cutedsl/test_grouped_swiglu_sfd_tiles.py",
        "test/python/gemm/cutedsl/test_swiglu_amax_per_call.py",
        "test/python/gemm/cutedsl/test_swiglu_wrapper_launch_stream.py",
        "test/python/gemm/cutedsl/test_grouped_gemm_swiglu.py",
        "test/python/gemm/cutedsl/test_grouped_gemm_canonical_layouts.py",
    ):
        selected["runtime/" + relative] = source / relative
    manifest = {
        "stage": "FULL24_PRIVATE_C_STORE_PARITY_AUDITED_QUALITY_STILL_FAILED",
        "scope": "All24 production-arithmetic single-component MXFP8 native configuration parity,9complete cases/216native calls/12 saved complete-logit tensors independentlyCPUaudited; unchanged-original/protectedemit/protectedomit byteidentical, original1%/5% gates stillfail. No quality/genKV/performance/publicAPI acceptance.",
        "runs": runs,
        "files": [{"path": name, "size": path.stat().st_size, "sha256": sha(path)} for name, path in sorted(selected.items())],
        "excluded": ["models", "cache", "engines", ".pt outputs", "native binaries"],
    }
    destination = ROOT / "results/a3-production-c-store-full24-pair-archive-20261001"
    destination.mkdir(exist_ok=False)
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for name, path in selected.items():
            tar.add(path, arcname=PREFIX + "/" + name, recursive=False)
        payload = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    record = {"archive": str(archive), "sha256": sha(archive), "size": archive.stat().st_size, "files": len(selected) + 1}
    (destination / "transport.json").write_text(json.dumps(record, indent=2) + "\n")
    print(json.dumps(record), flush=True)


if __name__ == "__main__":
    main()
