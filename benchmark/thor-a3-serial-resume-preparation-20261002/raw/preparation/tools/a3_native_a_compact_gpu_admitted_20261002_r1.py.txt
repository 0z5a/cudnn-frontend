"""Separate compact-quantizer GPU process under a verified parent shared flock."""

import argparse
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import sys

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
HEAD = "b57c65f1a1b60c2405e1e04c6b5ec6c6b2903bb5"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", required=True)
    args = p.parse_args()
    holder, fd = (int(os.environ[name]) for name in ("A3_ITERATION_LOCK_HOLDER", "A3_ITERATION_LOCK_FD"))
    link = Path(f"/proc/{holder}/fd/{fd}")
    assert link.stat().st_ino == Path("/tmp/codex-thor-perf.lock").stat().st_ino
    assert "FLOCK ADVISORY WRITE" in " ".join(Path(f"/proc/{holder}/fdinfo/{fd}").read_text().split())
    assert "torch" not in sys.modules and os.environ["CUDA_VISIBLE_DEVICES"] == "0" and os.environ["CUDA_MODULE_LOADING"] == "EAGER"
    native = json.loads((ROOT / "results/a3-latest-b1fe-native-provenance-20261002.json").read_text())
    assert native["head"] == HEAD and native["native_build"] == "FRESH_RC0"
    assert sha(native["binding"]["path"]) == native["binding"]["sha256"]
    assert all(sha(Path(native["source"]) / row["path"]) == row["sha256"] for row in native["native_input_digests"])
    assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
    args.expected_head, args.expected_source, args.expected_binding_sha = HEAD, native["source"], native["binding"]["sha256"]
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    (out / "admission.json").write_text(json.dumps({"stage": "VERIFIED_PARENT_FLOCK_BEFORE_TORCH", "holder": holder, "fd": fd, "head": HEAD}) + "\n")
    tools = Path(__file__).parent
    q = load("a3_guard_boundary_quant", tools / "a3_mxfp8_bf16_staged_full_model_worker_20261001.py")
    q.t.set_num_threads(4)
    q.t.backends.cuda.matmul.allow_tf32 = q.t.backends.cudnn.allow_tf32 = False
    q.t.set_float32_matmul_precision("highest")
    assert q.t.cuda.get_device_capability() == (11, 0)
    for imported in (q.m.cudnn, q.m.api, q.m._compiled_module):
        assert Path(imported.__file__).resolve().is_relative_to(Path(native["source"]) / "python")
    assert sha(q.m._compiled_module.__file__) == native["binding"]["sha256"]
    h = load("a3_guard_boundary_cuda", tools / "a3_guarded_rounding_native_20261002_r7.py")
    assert h.NAME.endswith("_r7")
    worker = load("a3_compact_native_A_complete_pair", tools / "a3_native_a_compact_full_model_generation_pair_20261002_r1.py")
    e2e = out / "e2e"
    e2e.mkdir()
    worker.execute(args, e2e, ROOT / "results/a3-weight-plan-20261002-r1/candidate/boundaries")
    child = json.loads((e2e / "result.json").read_text())
    assert child["quality_all_five_numeric_gates_pass"]
    assert child["native_main_calls"] == child["native_repair_calls"] == 13056 and child["native_postprocess_kernel_launches"] == 39168
    assert child["full_forward_probe"]["counts"] == [144, 144, 432]
    assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
    assert all(sha(Path(native["source"]) / row["path"]) == row["sha256"] for row in native["native_input_digests"])
    result = {
        "stage": "COMPACT_NATIVE_A_LATEST_GPU_COMPLETE_PENDING_INDEPENDENT_CPU_AUDIT",
        "head": HEAD,
        "binding": native["binding"],
        "child_result_sha256": sha(e2e / "result.json"),
        "total_native_main_calls": 13200,
        "total_native_repair_calls": 13200,
        "total_postprocess_kernel_launches": 39600,
        "accepted_production_optimization": False,
        "service_final": "inactive",
    }
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps(result), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
