"""CPU-only complete gate/up checkpoint and saved-input residual diagnostics.

Independent logical SF32 quantizer. No producer, cuDNN or model imports.
Two/three components are reconstruction controls, not native recipe success.
"""

import argparse
import hashlib
import json
import os
from pathlib import Path

os.environ["CUDA_VISIBLE_DEVICES"] = ""
import torch as t
from safetensors import safe_open

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
MODEL = Path("/home/jwipc/models/granite-3.1-1b-a400m-instruct-0da7a48b/model.safetensors")
WEIGHT_SHA = "ac02591061f1344027a7e7b11dbb4143f75f166c47dc09b742f5de3ab1dde1d1"


def sha_file(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for b in iter(lambda: f.read(4 * 1024 * 1024), b""):
            h.update(b)
    return h.hexdigest()


def reconstructed_component(residual):
    blocks = residual.reshape(-1, 32)
    peak = blocks.abs().amax(-1)
    exponent = t.where(peak > 0, t.ceil(t.log2(peak * (1 / 448))), 0).clamp(-126, 127)
    sf = t.pow(2.0, exponent)
    q = (blocks / sf[:, None]).clamp(-448, 448).to(t.float8_e4m3fn)
    return (q.float() * sf[:, None]).reshape(residual.shape)


def diagnose(value):
    original = value.float()
    effective = t.zeros_like(original)
    rows = []
    for component in range(1, 4):
        effective += reconstructed_component(original - effective)
        if component >= 2:
            delta = effective - original
            rows.append(
                {
                    "components": component,
                    "elements": value.numel(),
                    "max_abs": delta.abs().max().item(),
                    "relative_l2_fp64": (t.linalg.vector_norm(delta.double()) / t.linalg.vector_norm(original.double()).clamp_min(1e-12)).item(),
                    "fp32_elements_differ": (effective != original).sum().item(),
                    "bf16_recast_elements_differ": (effective.to(t.bfloat16) != value).sum().item(),
                }
            )
    return rows


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    t.set_num_threads(4)
    assert not t.cuda.is_initialized() and sha_file(MODEL) == WEIGHT_SHA
    weights, inputs = [], []
    with t.inference_mode(), safe_open(str(MODEL), framework="pt", device="cpu") as f:
        for layer in range(24):
            name = f"model.layers.{layer}.block_sparse_moe.input_linear.weight"
            value = f.get_tensor(name)
            assert value.shape == (32, 1024, 1024) and value.dtype == t.bfloat16
            row = {"layer": layer, "checkpoint_key": name, "rows": diagnose(value)}
            weights.append(row)
            print(json.dumps({"weights": row}), flush=True)
            (out / "partial.json").write_text(json.dumps({"weights": weights, "inputs": inputs}, indent=2))
        trace_path = ROOT / "results/a3-all24-precision-probe-20261001-r1/outputs.pt"
        traces = t.load(trace_path, weights_only=True, map_location="cpu")
        for backend in ("native", "reference"):
            for layer in range(24):
                value = traces["baseline_" + backend][layer]["x"]
                assert value.shape == (73, 1024) and value.dtype == t.bfloat16
                row = {"backend": backend, "layer": layer, "rows": diagnose(value)}
                inputs.append(row)
        assert not t.cuda.is_initialized()
        result = {
            "stage": "CPU_COMPONENT_DIAGNOSTIC_COMPLETE_NOT_NATIVE_PASS",
            "scope": __doc__,
            "cuda_initialized": False,
            "weight_sha256": WEIGHT_SHA,
            "checkpoint_gate_up_elements": sum(x["rows"][0]["elements"] for x in weights),
            "weights": weights,
            "inputs": inputs,
            "input_trace_sha256": sha_file(trace_path),
            "harness_sha256": sha_file(__file__),
            "native_calls": 0,
            "formal_performance": "NOT_RUN",
            "scope_limit": "Independent CPU logical quantization of checkpoint rows and earlier single-component trajectories; no GPU component/pack parity or all24 accuracy pass claimed",
        }
        assert result["checkpoint_gate_up_elements"] == 805306368
        (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
        print(json.dumps({"stage": result["stage"], "weights": len(weights), "input_traces": len(inputs)}), flush=True)


if __name__ == "__main__":
    main()
