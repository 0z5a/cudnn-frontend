"""Archive selected immutable residual/staging evidence; exclude tensor payloads."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-residual-staging-evidence-20261001"
NAMES = (
    "a3-residual-staging-20261001-r1",
    "a3-residual-staging-cpu-audit-20261001-r1",
    "a3-component-reconstruction-cpu-20261001-r1",
    "a3-bf16-rounding-cpu-20261001-r2",
    "a3-bf16-rounding-cpu-audit-20261001-r2",
)


def sha_file(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for b in iter(lambda: f.read(4 * 1024 * 1024), b""):
            h.update(b)
    return h.hexdigest()


def main():
    destination = ROOT / "results/a3-residual-staging-archive-20261001"
    destination.mkdir(exist_ok=False)
    selected = {}
    for name in NAMES:
        launch = ROOT / "results" / (name + "-launch")
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] == 0
        for folder in (ROOT / "results" / name, launch):
            for path in sorted(folder.rglob("*")):
                if path.is_file() and path.suffix in (".json", ".log"):
                    selected[str(path.relative_to(ROOT))] = path
    failed = "a3-bf16-rounding-boundaries-20261001-r1"
    failed_launch = ROOT / "results" / (failed + "-launch")
    timeout = json.loads((failed_launch / "run.json").read_text())
    assert timeout["status"] == "TIMEOUT" and timeout["returncode"] == -15
    for folder in (ROOT / "results" / failed, failed_launch):
        for path in sorted(folder.rglob("*")):
            if path.is_file() and path.suffix in (".json", ".log"):
                selected[str(path.relative_to(ROOT))] = path
    for name in (
        "diagnose_a3_residual_staging_20261001.py",
        "audit_a3_residual_staging_20261001.py",
        "diagnose_a3_component_reconstruction_cpu_20261001.py",
        "diagnose_a3_bf16_rounding_boundaries_20261001.py",
        "audit_a3_bf16_rounding_boundaries_20261001.py",
        "diagnose_a3_bf16_rounding_cpu_r2_20261001.py",
        "audit_a3_bf16_rounding_cpu_r2_20261001.py",
        "archive_a3_residual_staging_20261001.py",
        "a3_model_e2e_mergefix_20261001.py",
        "audit_a3_precision_followup_r2_20261001.py",
    ):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    records = [{"path": name, "origin": str(path), "size": path.stat().st_size, "sha256": sha_file(path)} for name, path in sorted(selected.items())]
    for path in selected.values():
        assert path.is_file() and not path.is_symlink() and path.suffix in (".json", ".log", ".py")
    manifest = {
        "stage": "RAW_RESIDUAL_STAGING_EVIDENCE_SELECTED",
        "tested_runtime_head": "7bfeda6bfeaa63a1e59fadc5d9fb48676f03f744",
        "files": records,
        "excluded": ["models", "cache", "engines", ".pt outputs", "native binaries", "object files"],
        "scope": "Torch causal controls and CPU audits; no new native recipe or production code accepted",
        "gpu_flag_control": "NOT_RUN_SHARED_LOCK_TIMEOUT",
        "preserved_lock_timeout": timeout,
        "unexecuted_harness": "tools/audit_a3_bf16_rounding_boundaries_20261001.py (GPU flag result absent)",
    }
    payload = (json.dumps(manifest, indent=2) + "\n").encode()
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for record in records:
            tar.add(selected[record["path"]], arcname=PREFIX + "/" + record["path"], recursive=False)
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    transport = {
        "archive": str(archive),
        "size": archive.stat().st_size,
        "sha256": sha_file(archive),
        "files": len(records) + 1,
        "payload_bytes": sum(x["size"] for x in records) + len(payload),
    }
    (destination / "transport.json").write_text(json.dumps(transport, indent=2) + "\n")
    print(json.dumps(transport), flush=True)


if __name__ == "__main__":
    main()
