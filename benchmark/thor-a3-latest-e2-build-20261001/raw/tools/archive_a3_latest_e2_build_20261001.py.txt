"""Archive completed latest-head fresh build evidence; no pending GPU result."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-latest-e2-build-evidence-20261001"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    selected = {}
    for name in ("a3-latest-e2-prepare-build-20261001-r1-launch", "build-a3-latest-e2-20261001-r1"):
        directory = ROOT / "results" / name
        run = json.loads((directory / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] == 0 and run["signal"] is None
        for path in directory.iterdir():
            if path.is_file() and path.suffix in (".json", ".log"):
                selected[str(path.relative_to(ROOT))] = path
    for name in ("a3-latest-e2-prebuild-20261001.json", "a3-latest-e2-native-provenance-20261001.json"):
        path = ROOT / "results" / name
        selected[str(path.relative_to(ROOT))] = path
    provenance = json.loads((ROOT / "results/a3-latest-e2-native-provenance-20261001.json").read_text())
    assert provenance["native_build"] == "FRESH_RC0"
    assert sha(Path(provenance["binding"]["path"])) == provenance["binding"]["sha256"]
    for name in ("prepare_build_a3_latest_e2_20261001.py", "archive_a3_latest_e2_build_20261001.py"):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    manifest = {
        "stage": "LATEST_HEAD_FRESH_NATIVE_BUILD_RC0_GPU_REGRESSION_NOT_RUN",
        "head": provenance["head"],
        "base": provenance["base"],
        "files": [{"path": name, "size": path.stat().st_size, "sha256": sha(path)} for name, path in sorted(selected.items())],
        "excluded": [".pt", "models", "cache", "engines", "native binaries", "pending GPU regression result"],
    }
    destination = ROOT / "results/a3-latest-e2-build-archive-20261001"
    destination.mkdir(exist_ok=False)
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for name, path in selected.items():
            tar.add(path, arcname=PREFIX + "/" + name, recursive=False)
        payload = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    record = {"archive": str(archive), "sha256": sha(archive), "size": archive.stat().st_size, "files": len(selected) + 1}
    (destination / "transport.json").write_text(json.dumps(record, indent=2) + "\n")
    print(json.dumps(record), flush=True)


if __name__ == "__main__":
    main()
