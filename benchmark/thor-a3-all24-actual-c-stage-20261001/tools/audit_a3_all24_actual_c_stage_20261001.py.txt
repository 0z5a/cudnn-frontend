"""Independent CPU audit of native C/epilogue isolation; no new GPU or model calls."""

import argparse
import importlib.util
import json
import os
from pathlib import Path
import subprocess

os.environ["CUDA_VISIBLE_DEVICES"] = ""
HELPER = Path(__file__).with_name("audit_a3_precision_followup_r2_20261001.py")
spec = importlib.util.spec_from_file_location("all24_stage_cpu", HELPER)
a = importlib.util.module_from_spec(spec)
spec.loader.exec_module(a)
t = a.torch
HEAD = "741e10ffa5e93dbc0a8a41e06e66e0c81c1f85b1"


def cpu_stage(c, fp16):
    split = c.to(t.bfloat16).reshape(584, -1, 2, 32)
    stage = (t.nn.functional.silu(split[:, :, 0]) * split[:, :, 1]).reshape(584, 512).float()
    if not fp16:
        return stage
    blocks = stage.reshape(584, -1, 32)
    maximum = blocks.abs().amax(-1)
    mantissa, exponent = t.frexp(maximum)
    exponent = t.where(maximum > 0, exponent - (mantissa == 0.5).int(), 0).clamp(-126, 127)
    scale = t.ldexp(t.ones_like(maximum), exponent)
    return ((blocks / scale.unsqueeze(-1)).to(t.float16).float() * scale.unsqueeze(-1)).reshape(584, 512)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    t.set_num_threads(4)
    prior = a.ROOT / "results/a3-residual-staging-20261001-r1"
    captured = t.load(prior / "outputs.pt", map_location="cpu", weights_only=True)
    directory = a.ROOT / "results/a3-all24-actual-c-stage-20261001-r2"
    meta = json.loads((directory / "result.json").read_text())
    run = json.loads((directory.with_name(directory.name + "-launch") / "run.json").read_text())
    assert run["status"] == "EXITED" and run["returncode"] == 0 and run["signal"] is None
    assert meta["source_head"] == HEAD and meta["native_calls"] == 48 and not meta["original_baseline_replaced"]
    assert meta["model_parameters"] == 1334628352 and meta["checkpoint_sha256"] == "ac02591061f1344027a7e7b11dbb4143f75f166c47dc09b742f5de3ab1dde1d1"
    assert meta["formal_performance"] == "NOT_RUN" and meta["service_final"] == "inactive"
    source = Path(meta["source"])
    assert subprocess.check_output(["git", "-C", str(source), "rev-parse", "HEAD"], text=True).strip() == HEAD
    assert not subprocess.check_output(["git", "-C", str(source), "status", "--porcelain", "--untracked-files=no"], text=True).strip()
    for record in meta["imports"].values():
        assert Path(record["path"]).resolve().is_relative_to(source / "python") and a.sha_file(record["path"]) == record["sha256"]
    native = json.loads((a.ROOT / "results/a3-latest-e2-native-provenance-20261001.json").read_text())
    assert native["native_build"] == "FRESH_RC0" and native["binding"]["sha256"] == meta["imports"]["binding"]["sha256"]
    assert list(meta["native_plans"].values()) == [meta["prototype"]]
    assert a.sha_file(meta["prototype"]["path"]) == meta["prototype"]["sha256"]
    assert a.sha_file(run["argv"][1]) == meta["harness_sha256"]
    assert a.sha_file(a.ROOT / "tools/a3_double_mxfp8_recipe_probe_20261001.py") == meta["helper_sha256"]
    assert a.sha_file(prior / "outputs.pt") == meta["prior_outputs_sha256"] and a.sha_file(prior / "result.json") == meta["prior_metadata_sha256"]
    assert a.sha_file(directory / "outputs.pt") == meta["outputs_sha256"]
    values = t.load(directory / "outputs.pt", map_location="cpu", weights_only=True)
    assert len(values) == 288 and len(meta["rows"]) == 48
    records = []
    for layer in range(24):
        original_c = captured[f"local/{layer}"]["original_bf16_local"]["c"]
        original_d = captured[f"local/{layer}"]["original_bf16_local"]["d"]
        counts = t.bincount(captured[f"stock_route/{layer}"]["index"].flatten(), minlength=32).tolist()
        for dtype in ("torch.bfloat16", "torch.float16"):
            rows = [r for r in meta["rows"] if r["layer"] == layer and r["dtype"] == dtype]
            assert len(rows) == 1
            row = rows[0]
            prefix = f"{layer}/{dtype}/"
            assert row["native_calls"] == 1 and row["route_counts"] == counts and sum(counts) == 584
            assert row["padded_rows"] == sum((n + 255) // 256 * 256 for n in counts)
            c, d, expected, raw, sf, dest = [values[prefix + n] for n in ("c", "d", "expected_actual_c_gpu_stage", "raw_d", "raw_sfd", "destinations")]
            assert c.shape == (584, 1024) and d.shape == expected.shape == (584, 512) and str(raw.dtype) == dtype
            n, k = raw.shape
            assert n == row["padded_rows"] and k == 512
            r, col = t.arange(n)[:, None], t.arange(k // 32)[None, :]
            address = (((r // 128) * (k // 128) + col // 4) * 32 + r % 32) * 16 + ((r % 128) // 32) * 4 + col % 4
            code = sf.view(t.uint8).reshape(-1)[address]
            scale = t.ldexp(t.ones_like(code, dtype=t.float32), code.int() - 127)
            decoded = (raw.float().reshape(n, k // 32, 32) * scale.unsqueeze(-1)).reshape(n, k)[dest]
            assert t.equal(decoded.view(t.uint8), d.view(t.uint8))
            a.compare_metric(a.difference(d, expected, t.float32), row["native_d_actual_c_gpu_reference"])
            a.compare_metric(a.difference(d, original_d, t.float32), row["native_d_original_stock"])
            different = (d != expected).sum().item()
            assert different == row["native_d_actual_c_gpu_different_values"]
            c_different = (c.to(t.bfloat16) != original_c).sum().item()
            assert c_different == row["native_c_original_bf16_cast_mismatches"]
            recomputed = cpu_stage(c, dtype == "torch.float16")
            records.append(
                {
                    "layer": layer,
                    "dtype": dtype,
                    "native_D_raw_decode_byte_exact": True,
                    "native_actual_C_GPU_stage_different_values": different,
                    "native_actual_C_GPU_stage_fp64": a.difference(d, expected),
                    "native_original_D_fp64": a.difference(d, original_d),
                    "C_original_cast_different_values": c_different,
                    "CPU_actual_C_stage_vs_saved_GPU_different_values": (recomputed != expected).sum().item(),
                    "CPU_actual_C_stage_vs_saved_GPU_fp64": a.difference(recomputed, expected),
                    "CPU_actual_C_stage_native_fp64": a.difference(d, recomputed),
                    "route_counts": counts,
                }
            )
    original = a.ROOT / "results/a3-all24-actual-c-stage-20261001-r1-launch"
    original_run = json.loads((original / "run.json").read_text())
    assert original_run["returncode"] == 1
    assert "assert d.dtype == t.float16" in (original / "stderr.log").read_text()
    assert not t.cuda.is_initialized()
    result = {
        "stage": "CPU_AUDITED_ALL24_FIXED_INPUT_EPILOGUE_ISOLATION_NOT_MODEL_SUCCESS",
        "source_head": HEAD,
        "run": run,
        "preserved_r1_driver": original_run,
        "native_calls_observed": 48,
        "new_native_calls": 0,
        "cuda_initialized": False,
        "rows": records,
        "tensor_count": len(values),
        "tensors": a.flatten_tensors(values),
        "producer_outputs_sha256": meta["outputs_sha256"],
        "harness_sha256": a.sha_file(__file__),
        "cpu_helper_sha256": a.sha_file(HELPER),
        "formal_performance": "NOT_RUN",
        "service_final": "inactive",
        "limits": "24 fixed original input/route/weight trajectories, two native D storage modes; no full-model feedback/generation/KV/performance/accepted feature. CPU activation and GPU activation references separately compared, tiny normalized FP16 differences are recorded without being reclassified as a model pass. No independent new input-physical-pack capture or runtime compiled-plan identity proof.",
    }
    (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
    print(
        json.dumps(
            {
                "stage": result["stage"],
                "BF16_rows_with_D_GPU_stage_differences": sum(
                    r["native_actual_C_GPU_stage_different_values"] > 0 for r in records if r["dtype"] == "torch.bfloat16"
                ),
                "FP16_rows_with_D_GPU_stage_differences": sum(
                    r["native_actual_C_GPU_stage_different_values"] > 0 for r in records if r["dtype"] == "torch.float16"
                ),
            }
        ),
        flush=True,
    )


if __name__ == "__main__":
    main()
