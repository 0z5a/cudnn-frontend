"""After the existing cache generation: CPU audit, fresh build, latest31, new iteration.

One parent flock covers fresh compilation and serial child GPU phases, avoiding
CPU compilation during warmed request measurements. The existing cache waiter
is never duplicated or edited. Known historical/default quality and application
failures are retained. Nothing here accepts a production feature or completes A3.
"""

import argparse
import fcntl
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
HEAD = "b57c65f1a1b60c2405e1e04c6b5ec6c6b2903bb5"
LOGGER = "/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--out", required=True)
    args = p.parse_args()
    out = Path(args.out)
    out.mkdir(exist_ok=False)
    tools = Path(__file__).parent
    names = [
        Path(__file__).name,
        "a3_latestb1fe_gpu_phase_admitted_20261002_r1.py",
        "prepare_build_a3_latest_b1fe_20261002_r1.py",
        "run_a3_latest_b1fe_regression_queued_20261002_r1.py",
        "a3_integer_reduce_boundary_pair_model_20261002_r1.py",
        "a3_guarded_rounding_boundary_queued_20261002_r9.py",
        "a3_guarded_rounding_full_model_20261002_r9.py",
        "a3_guarded_rounding_native_20261002_r8.py",
        "a3_guarded_rounding_native_20261002_r7.py",
        "a3_guarded_rounding_model_runtime_20261002_r5.py",
        "a3_guarded_rounding_request_runtime_20261002_r4.py",
        "a3_integer_reduce_full_generation_pair_20261002_r1.py",
        "audit_a3_integer_reduce_boundary_cpu_20261002_r1.py",
        "audit_a3_integer_reduce_pair_cpu_20261002_r1.py",
        "audit_a3_integer_reduce_full_model_cpu_20261002_r1.py",
        "audit_a3_integer_reduce_generation_pair_cpu_20261002_r1.py",
        "audit_a3_weight_plan_generation_pair_cpu_20261002_r2.py",
        "a3_mxfp8_bf16_staged_full_model_worker_20261001.py",
        "a3_all24_precision_probe_20261001.py",
        "a3_model_e2e_mergefix_20261001.py",
        "a3_mxfp8_bf16_staged_native_kernel_20261001.py",
        "diagnose_a3_fixed_window_cpu_20261001.py",
    ]
    frozen = {str(tools / name): sha(tools / name) for name in names}
    bundle = tools / "a3-latest-b1fe-20261002.bundle"
    assert sha(bundle) == "27d265ba9420625d84ee6b048e61d120fc7a7cc4e43e7936353d10b01e6c4db3"
    frozen[str(bundle)] = sha(bundle)
    (out / "frozen_inputs.json").write_text(json.dumps({"head": HEAD, "files": frozen}, indent=2) + "\n")
    parent = ROOT / "results/a3-weight-plan-generation-20261002-r3-launch/run.json"
    start = time.monotonic()
    while not parent.exists():
        assert "torch" not in sys.modules
        if time.monotonic() - start > 6000:
            (out / "result.json").write_text(json.dumps({"stage": "NOT_RUN_CACHE_PARENT_NOT_TERMINAL", "torch_imported": False}) + "\n")
            return 75
        time.sleep(2)
    cached = json.loads(parent.read_text())
    assert cached["status"] == "EXITED" and cached["signal"] is None
    if cached["returncode"] != 0:
        (out / "result.json").write_text(json.dumps({"stage": "NOT_RUN_CACHE_PARENT_NONZERO", "parent_run": cached, "torch_imported": False}) + "\n")
        return cached["returncode"]
    old_python = ROOT / "venvs/a3sfdlatest29a0/bin/python"
    runs = []

    def logged(name, argv, python, gpu=False, timeout=1200):
        result = ROOT / "results" / name
        launch = result.with_name(name + "-launch")
        assert not launch.exists() and not result.exists()
        env = os.environ.copy()
        env.update(CUDA_VISIBLE_DEVICES="0" if gpu else "", CUDA_MODULE_LOADING="EAGER", PYTHONPATH=str(ROOT / "a3sfdlatestb1fe/python"))
        rc = subprocess.run(
            [str(python), LOGGER, "--out", str(launch), "--cwd", str(ROOT), "--timeout", str(timeout), "--", *map(str, argv)], env=env
        ).returncode
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] == rc and run["signal"] is None
        runs.append({"name": name, "run": run})
        (out / "completed_runs.json").write_text(json.dumps(runs, indent=2) + "\n")
        if rc:
            raise RuntimeError(f"Actual phase {name} failed rc{rc}; see preserved launch logs")
        return result

    logged(
        "a3-weight-plan-generation-pair-cpu-audit-20261002-r3",
        [
            old_python,
            tools / "audit_a3_weight_plan_generation_pair_cpu_20261002_r2.py",
            "--out",
            ROOT / "results/a3-weight-plan-generation-pair-cpu-audit-20261002-r3",
        ],
        old_python,
        timeout=180,
    )
    assert all(sha(path) == digest for path, digest in frozen.items())
    with open("/tmp/codex-thor-perf.lock", "a") as lock:
        begin = time.monotonic()
        while True:
            try:
                fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                break
            except BlockingIOError:
                if time.monotonic() - begin > 3600:
                    (out / "result.json").write_text(
                        json.dumps({"stage": "NOT_RUN_NEXT_SHARED_LOCK_BUSY", "cached_parent_audit_pass": True, "torch_imported": False}) + "\n"
                    )
                    return 75
                time.sleep(1)
        assert "torch" not in sys.modules and all(sha(path) == digest for path, digest in frozen.items())
        os.environ.update(A3_ITERATION_LOCK_HOLDER=str(os.getpid()), A3_ITERATION_LOCK_FD=str(lock.fileno()))
        (out / "admission.json").write_text(
            json.dumps({"stage": "LOCK_ADMITTED", "pid": os.getpid(), "wait_seconds": time.monotonic() - begin, "torch_imported": False}) + "\n"
        )
        logged("prepare-build-a3-latest-b1fe-20261002-r1", [old_python, tools / "prepare_build_a3_latest_b1fe_20261002_r1.py"], old_python, timeout=3600)
        python = ROOT / "venvs/a3sfdlatestb1fe/bin/python"
        driver = tools / "a3_latestb1fe_gpu_phase_admitted_20261002_r1.py"
        for phase, name in (("regression", "a3-latest-b1fe-regression-20261002-r1"), ("candidate", "a3-integer-reduction-latestb1fe-20261002-r1")):
            logged(name, [python, driver, "--phase", phase, "--out", ROOT / "results" / name], python, gpu=True)
        candidate = ROOT / "results/a3-integer-reduction-latestb1fe-20261002-r1"
        for kind in ("boundary", "pair", "full_model"):
            name = f"a3-integer-reduction-{kind}-cpu-audit-20261002-r1"
            argv = [python, tools / f"audit_a3_integer_reduce_{kind}_cpu_20261002_r1.py", "--out", ROOT / "results" / name]
            if kind == "full_model":
                argv += ["--input", candidate]
            folder = logged(name, argv, python, timeout=300)
            proof = json.loads((folder / "result.json").read_text())
            assert not proof["cuda_initialized"] and proof["new_native_calls"] == 0
            if kind == "full_model":
                assert proof["quality_all_three_cases_both_gates"]
        pair = json.loads((candidate / "candidate/pair.json").read_text())
        gain = all(
            next(row for row in pair["comparisons"] if row["case"] == name)["cached_shuffle_over_integer_redux_ratio"] > 1.02
            for name in ("saved_first_expert", "granite_decode_e32_n1024_top8")
        )
        if gain:
            name = "a3-integer-reduction-generation-latestb1fe-20261002-r1"
            logged(name, [python, driver, "--phase", "generation", "--out", ROOT / "results" / name], python, gpu=True, timeout=2400)
            name = "a3-integer-reduction-generation-pair-cpu-audit-20261002-r1"
            logged(name, [python, tools / "audit_a3_integer_reduce_generation_pair_cpu_20261002_r1.py", "--out", ROOT / "results" / name], python, timeout=300)
        assert all(sha(path) == digest for path, digest in frozen.items())
        assert subprocess.run(["systemctl", "is-active", "vllm.service"], capture_output=True, text=True).stdout.strip() == "inactive"
        result = {
            "stage": "LATEST_B1FE_INTEGER_REDUX_ITERATION_TERMINAL" if gain else "LATEST_B1FE_INTEGER_REDUX_NO_MEANINGFUL_WRAPPER_GAIN_REJECTED",
            "head": HEAD,
            "base": "b1fe8a0f9e7701cea8d6f7f0ad8c11d043e024fc",
            "production_gpu_tests_passed": 31,
            "cached_parent_independent_cpu_audit_pass": True,
            "integer_redux_candidate_independent_cpu_audits_pass": True,
            "integer_redux_cached_generation_and_whole_model_pairs": "COMPLETE_CPU_AUDITED" if gain else "NOT_RUN_REJECTED_WRAPPER_GAIN",
            "comparisons": pair["comparisons"],
            "runs": runs,
            "files": frozen,
            "production_feature_accepted": False,
            "original_default_k1024_quality": "HISTORICAL_FAILED_RECIPE_NO_CLAIM_FIXED",
            "original_passcode83_application_answer": "FALSE_UNCHANGED",
            "service_final": "inactive",
        }
        (out / "result.json").write_text(json.dumps(result, indent=2) + "\n")
        print(json.dumps(result), flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
