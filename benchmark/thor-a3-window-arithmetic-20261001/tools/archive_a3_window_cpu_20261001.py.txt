"""Archive raw CPU window logs/metadata/source, excluding tensors and models."""

import hashlib
import io
import json
from pathlib import Path
import tarfile

ROOT = Path("/home/jwipc/experiments/cudnn-sm110-a1-a3-20260928")
PREFIX = "a3-fixed-window-cpu-evidence-20261001"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    selected = {}
    names = [
        "a3-fixed-window-cpu-20261001-r1",
        "a3-fixed-window-all24-cpu-20261001-r1",
        "a3-mxfp8-window-cpu-20261001-r1",
        "a3-window-cpu-audit-20261001-r1",
        "a3-window-cpu-audit-20261001-r2",
    ]
    for name in names:
        directory = ROOT / "results" / name
        launch = directory.with_name(name + "-launch")
        run = json.loads((launch / "run.json").read_text())
        assert run["status"] == "EXITED" and run["returncode"] == 0 and run["signal"] is None
        result = json.loads((directory / "result.json").read_text())
        assert not result["cuda_initialized"] and result["native_calls"] == 0
        assert sha(Path(run["argv"][1])) == result["harness_sha256"]
        for base in (directory, launch):
            for path in sorted(base.rglob("*")):
                if path.is_file() and path.suffix in (".json", ".log", ".txt"):
                    selected[str(path.relative_to(ROOT))] = path
    for name in (
        "diagnose_a3_fixed_window_cpu_20261001.py",
        "validate_a3_fixed_window_all24_cpu_20261001.py",
        "validate_a3_mxfp8_window_cpu_20261001.py",
        "audit_a3_window_cpu_20261001.py",
        "audit_a3_window_cpu_r2_20261001.py",
        "archive_a3_window_cpu_20261001.py",
    ):
        path = ROOT / "tools" / name
        selected[str(path.relative_to(ROOT))] = path
    for name in (
        "a3-stock-dot-ncu-20261001-r3/target/result.json",
        "a3-bf16-dot-fp32-output-20261001-r1/result.json",
        "a3-all24-actual-c-stage-20261001-r2/result.json",
        "a3-residual-staging-20261001-r1/result.json",
        "a3-stock-dot-ncu-analysis-20261001-r3/result.json",
        "a3-stock-dot-ncu-analysis-20261001-r3/sass.txt",
        "a3-latest-e2-native-provenance-20261001.json",
    ):
        path = ROOT / "results" / name
        selected[str(path.relative_to(ROOT))] = path
    selected["tools/run_logged.py"] = Path("/home/jwipc/experiments/sm110-validation-round2-20260928/pack/scripts/run_logged.py")
    manifest = {
        "stage": "CPU_WINDOW_ARITHMETIC_DIAGNOSIS_AUDITED_NOT_MODEL_SUCCESS",
        "files": [{"path": name, "size": path.stat().st_size, "sha256": sha(path)} for name, path in sorted(selected.items())],
        "excluded": ["models", "cache", "engines", ".pt outputs", "native binaries", ".ncu-rep"],
        "scope": "CPU predictions reproduce frozen BF16 and actual MXFP8 firstexpertFP32; all24 sampledBF16 heldout. Full-model precision remains failed. Five terminal rc0 CPU logs and160 independently byte/hash-audited saved tensors, no new GPU/native calls.",
    }
    destination = ROOT / "results/a3-window-cpu-archive-20261001"
    destination.mkdir(exist_ok=False)
    archive = destination / (PREFIX + ".tar.gz")
    with tarfile.open(archive, "w:gz") as tar:
        for name, path in selected.items():
            tar.add(path, arcname=PREFIX + "/" + name, recursive=False)
        payload = (json.dumps(manifest, indent=2) + "\n").encode()
        info = tarfile.TarInfo(PREFIX + "/manifest.json")
        info.size = len(payload)
        tar.addfile(info, io.BytesIO(payload))
    record = {"archive": str(archive), "sha256": sha(archive), "size": archive.stat().st_size, "files": len(selected) + 1}
    (destination / "transport.json").write_text(json.dumps(record, indent=2) + "\n")
    print(json.dumps(record), flush=True)


if __name__ == "__main__":
    main()
